    46→// ===== Image Lightbox (Portal) =====
    47→function ImageLightbox({
    48→  isOpen, imageUrl, label, onClose
    49→}: {
    50→  isOpen: boolean
    51→  imageUrl: string
    52→  label: string
    53→  onClose: () => void
    54→}) {
    55→  const [zoom, setZoom] = useState(1)
    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→  const [isDragging, setIsDragging] = useState(false)
    58→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    59→  const [imageLoaded, setImageLoaded] = useState(false)
    60→  const containerRef = useRef<HTMLDivElement>(null)
    61→  const lastTouchDist = useRef<number | null>(null)
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) return
    70→    const handleKeyDown = (e: KeyboardEvent) => {
    71→      if (e.key === 'Escape') onClose()
    72→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    73→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    74→      else if (e.key === '0') resetView()
    75→    }
    76→    window.addEventListener('keydown', handleKeyDown)
    77→    return () => window.removeEventListener('keydown', handleKeyDown)
    78→  }, [isOpen, onClose, resetView])
    79→
    80→  const handleWheel = useCallback((e: React.WheelEvent) => {
    81→    e.preventDefault()
    82→    setZoom(z => {
    83→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    84→      if (next <= 1) setPosition({ x: 0, y: 0 })
    85→      return next
    86→    })
    87→  }, [])
    88→
    89→  const handleMouseDown = (e: React.MouseEvent) => {
    90→    if (zoom <= 1) return
    91→    setIsDragging(true)
    92→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    93→  }
    94→
    95→  const handleMouseMove = (e: React.MouseEvent) => {
    96→    if (!isDragging) return
    97→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
    98→  }
    99→
   100→  const handleMouseUp = () => setIsDragging(false)
   101→
   102→  const handleTouchStart = (e: React.TouchEvent) => {
   103→    if (e.touches.length === 2) {
   104→      const dx = e.touches[0].clientX - e.touches[1].clientX
   105→      const dy = e.touches[0].clientY - e.touches[1].clientY
   106→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   107→    } else if (e.touches.length === 1 && zoom > 1) {
   108→      setIsDragging(true)
   109→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   110→    }
   111→  }
   112→
   113→  const handleTouchMove = (e: React.TouchEvent) => {
   114→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   115→      const dx = e.touches[0].clientX - e.touches[1].clientX
   116→      const dy = e.touches[0].clientY - e.touches[1].clientY
   117→      const dist = Math.sqrt(dx * dx + dy * dy)
   118→      const scale = dist / lastTouchDist.current
   119→      setZoom(z => {
   120→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   121→        if (next <= 1) setPosition({ x: 0, y: 0 })
   122→        return next
   123→      })
   124→      lastTouchDist.current = dist
   125→    } else if (e.touches.length === 1 && isDragging) {
   126→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   127→    }
   128→  }
   129→
   130→  const handleTouchEnd = () => {
   131→    lastTouchDist.current = null
   132→    setIsDragging(false)
   133→  }
   134→
   135→  if (!isOpen) return null
   136→
   137→  return createPortal(
   138→    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col items-center justify-center">
   139→      <div
   140→        ref={containerRef}
   141→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   142→        onWheel={handleWheel}
   143→        onMouseDown={handleMouseDown}
   144→        onMouseMove={handleMouseMove}
   145→        onMouseUp={handleMouseUp}
   146→        onMouseLeave={handleMouseUp}
   147→        onTouchStart={handleTouchStart}
   148→        onTouchMove={handleTouchMove}
   149→        onTouchEnd={handleTouchEnd}
   150→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   151→      >
   152→        {!imageLoaded && (
   153→          <Loader2 className="size-10 text-white animate-spin" />
   154→        )}
   155→        <img
   156→          src={imageUrl}
   157→          alt={label}
   158→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   159→          style={{
   160→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   161→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   162→            display: imageLoaded ? 'block' : 'none',
   163→          }}
   164→          onLoad={() => setImageLoaded(true)}
   165→          draggable={false}
   166→        />
   167→      </div>
   168→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   169→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   170→          <ZoomOut className="size-5" />
   171→        </Button>
   172→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   173→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   174→          <ZoomIn className="size-5" />
   175→        </Button>
   176→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   177→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   178→        </Button>
   179→        <span className="text-white/70 text-sm ml-2">{label}</span>
   180→      </div>
   181→    </div>,
   182→    document.body
   183→  )
   184→}

   186→// ===== Exercise Detail Dialog =====
   187→function ExerciseDetailDialog({
   188→  exercise,
   189→  chapter,
   190→  onClose,
   191→}: {
   192→  exercise: ExerciseData
   193→  chapter: ChapterData
   194→  onClose: () => void
   195→}) {
   196→  const { user, chapterExercises, setSelectedExercise, setChapterExercises } = useAppStore()
   197→  const student = user as StudentUser
   198→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   199→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   200→  const [saving, setSaving] = useState(false)
   201→  const [comments, setComments] = useState<CommentData[]>([])
   202→  const [newComment, setNewComment] = useState('')
   203→  const [sendingComment, setSendingComment] = useState(false)
   204→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   205→
   206→  const currentIndex = chapterExercises.findIndex(e => e.id === exercise.id)
   207→  const totalExercises = chapterExercises.length
   208→
   209→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   210→
   211→  useEffect(() => {
   212→    let cancelled = false
   213→    if (commentsLoaded) return
   214→    const load = async () => {
   215→      try {
   216→        const res = await fetch('/api/comments')
   217→        if (res.ok && !cancelled) {
   218→          const data = await res.json()
   219→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   220→        }
   221→      } catch { /* ignore */ }
   222→      if (!cancelled) setCommentsLoaded(true)
   223→    }
   224→    load()
   225→  }, [exercise.id, commentsLoaded])
   226→
   227→  const refetchComments = async () => {
   228→    try {
   229→      const res = await fetch('/api/comments')
   230→      if (res.ok) {
   231→        const data = await res.json()
   232→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   233→      }
   234→    } catch { /* ignore */ }
   235→  }
   236→
   237→  const handleSave = async () => {
   238→    setSaving(true)
   239→    try {
   240→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   241→        method: 'PUT',
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   244→      })
   245→      toast.success('Progression sauvegardée !')
   246→      const updatedExercises = chapterExercises.map(e =>
   247→        e.id === exercise.id ? { ...e, progress: { id: e.progress?.id || '', status, studentNote: note } } : e
   248→      )
   249→      setChapterExercises(updatedExercises)
   250→    } catch {
   251→      toast.error('Erreur lors de la sauvegarde.')
   252→    }
   253→    setSaving(false)
   254→  }
   255→
   256→  const handleSendComment = async () => {
   257→    if (!newComment.trim()) return
   258→    setSendingComment(true)
   259→    try {
   260→      await fetch('/api/comments', {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   264→      })
   265→      setNewComment('')
   266→      refetchComments()
   267→    } catch {
   268→      toast.error("Erreur lors de l'envoi.")
   269→    }
   270→    setSendingComment(false)
   271→  }
   272→
   273→  const goToExercise = (direction: 'prev' | 'next') => {
   274→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   275→    if (newIndex < 0 || newIndex >= totalExercises) return
   276→    setSelectedExercise(chapterExercises[newIndex])
   277→  }
   278→
   279→  const imageUrl = exercise.pageStart ? exerciseImageUrl(chapter.number, exercise.number) : ''
   280→
   281→  return (
   282→    <>
   283→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   284→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   285→          <DialogHeader>
   286→            <DialogTitle className="text-lg">
   287→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   288→            </DialogTitle>
   289→            <DialogDescription className="sr-only">
   290→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   291→            </DialogDescription>
   292→          </DialogHeader>
   293→
   294→          {imageUrl && (
   295→            <div className="space-y-2">
   296→              <div className="relative rounded-lg overflow-hidden border bg-gray-50 max-h-64">
   297→                <img src={imageUrl} alt={`Exercice ${exercise.number}`} className="w-full h-full object-contain max-h-64" />
   298→              </div>
   299→              <Button variant="outline" size="sm" onClick={() => setLightbox({ isOpen: true, imageUrl, label: `Ex.${exercise.number}` })}>
   300→                <Eye className="size-4 mr-1" /> Agrandir
   301→              </Button>
   302→            </div>
   303→          )}
   304→
   305→          <div className="space-y-2">
   306→            <Label className="text-sm font-medium">Statut</Label>
   307→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   308→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   309→                <RadioGroupItem value="not_started" />
   310→                <Circle className="size-4 text-gray-500" />
   311→                <span className="text-sm text-gray-500">Non commencé</span>
   312→              </label>
   313→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   314→                <RadioGroupItem value="in_progress" />
   315→                <Clock className="size-4 text-amber-600" />
   316→                <span className="text-sm text-amber-700">En cours</span>
   317→              </label>
   318→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   319→                <RadioGroupItem value="completed" />
   320→                <CheckCircle className="size-4 text-emerald-600" />
   321→                <span className="text-sm text-emerald-700">Terminé</span>
   322→              </label>
   323→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   324→                <RadioGroupItem value="difficulty" />
   325→                <AlertTriangle className="size-4 text-red-600" />
   326→                <span className="text-sm text-red-700">Difficulté</span>
   327→              </label>
   328→            </RadioGroup>
   329→          </div>
   330→
   331→          <div className="space-y-2">
   332→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   333→            <Textarea
   334→              id="notes"
   335→              value={note}
   336→              onChange={(e) => setNote(e.target.value)}
   337→              placeholder="Ajoutez vos notes ici..."
   338→              rows={3}
   339→            />
   340→          </div>
   341→
   342→          <div className="flex justify-end">
   343→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   344→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   345→              Sauvegarder
   346→            </Button>
   347→          </div>
   348→
   349→          <Separator />
   350→
   351→          <div className="space-y-3">
   352→            <h4 className="font-medium text-sm">Commentaires</h4>
   353→            {comments.map((comment) => (
   354→              <div key={comment.id} className="space-y-2">
   355→                <div className="flex items-start gap-2">
   356→                  <Avatar className="size-8">
   357→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   358→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   359→                    </AvatarFallback>
   360→                  </Avatar>
   361→                  <div className="flex-1">
   362→                    <div className="flex items-center gap-2">
   363→                      <span className="text-sm font-medium">
   364→                        {comment.student.firstName} {comment.student.lastName}
   365→                      </span>
   366→                      <span className="text-xs text-muted-foreground">
   367→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   368→                      </span>
   369→                    </div>
   370→                    <p className="text-sm mt-1">{comment.content}</p>
   371→                  </div>
   372→                </div>
   373→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   374→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   375→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   376→                    <span className="text-xs text-emerald-600 mt-1 block">
   377→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   378→                    </span>
   379→                  </div>
   380→                ))}
   381→              </div>
   382→            ))}
   383→            <div className="flex gap-2">
   384→              <Textarea
   385→                value={newComment}
   386→                onChange={(e) => setNewComment(e.target.value)}
   387→                placeholder="Ajouter un commentaire..."
   388→                rows={2}
   389→              />
   390→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   391→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   392→              </Button>
   393→            </div>
   394→          </div>
   395→
   396→          <DialogFooter className="flex-row justify-between">
   397→            <Button
   398→              variant="outline"
   399→              onClick={() => goToExercise('prev')}
   400→              disabled={currentIndex <= 0}
   401→            >
   402→              <ChevronLeft className="size-4 mr-1" /> Précédent
   403→            </Button>
   404→            <Button
   405→              variant="outline"
   406→              onClick={() => goToExercise('next')}
   407→              disabled={currentIndex >= totalExercises - 1}
   408→            >
   409→              Suivant <ChevronRight className="size-4 ml-1" />
   410→            </Button>
   411→          </DialogFooter>
   412→        </DialogContent>
   413→      </Dialog>
   414→
   415→      <ImageLightbox
   416→        key={lightbox.imageUrl}
   417→        isOpen={lightbox.isOpen}
   418→        imageUrl={lightbox.imageUrl}
   419→        label={lightbox.label}
   420→        onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   421→      />
   422→    </>
   423→  )
   424→}

   468→      toast.error('Erreur de connexion.')

   471→  }

   474→    setLoading(true)

   476→      const res = await fetch('/api/student/register', {
   477→        method: 'POST',
   478→        headers: { 'Content-Type': 'application/json' },
   479→        body: JSON.stringify({
   480→          firstName: regFirstName, lastName: regLastName, email: regEmail,
   481→          className: regClassName, schoolName: regSchoolName, phone: regPhone, password: regPassword,
   482→        }),

   485→        toast.success('Votre inscription a été envoyée. Elle sera validée par l\'enseignant.')
   486→        setMode('login')
   487→      } else {
   488→        const data = await res.json().catch(() => ({}))
   489→        toast.error(data.error || "Erreur lors de l'inscription.")

   492→      toast.error("Erreur lors de l'inscription.")
   493→    }
   494→    setLoading(false)
   495→  }

   497→  const handleTeacherLogin = async () => {
   498→    setLoading(true)
   499→    try {
   500→      const res = await fetch('/api/teacher/login', {
