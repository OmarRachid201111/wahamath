    45→// ===== Image Lightbox (Portal) =====
    46→function ImageLightbox({
    47→  isOpen, imageUrl, label, onClose
    48→}: {
    49→  isOpen: boolean
    50→  imageUrl: string
    51→  label: string
    52→  onClose: () => void
    53→}) {
    54→  const [zoom, setZoom] = useState(1)
    55→  const [position, setPosition] = useState({ x: 0, y: 0 })
    56→  const [isDragging, setIsDragging] = useState(false)
    57→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    58→  const [imageLoaded, setImageLoaded] = useState(false)
    59→  const containerRef = useRef<HTMLDivElement>(null)
    60→  const lastTouchDist = useRef<number | null>(null)
    61→
    62→  const resetView = useCallback(() => {
    63→    setZoom(1)
    64→    setPosition({ x: 0, y: 0 })
    65→  }, [])
    66→
    67→  useEffect(() => {
    68→    if (!isOpen) return
    69→    const handleKeyDown = (e: KeyboardEvent) => {
    70→      if (e.key === 'Escape') onClose()
    71→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    72→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    73→      else if (e.key === '0') resetView()
    74→    }
    75→    window.addEventListener('keydown', handleKeyDown)
    76→    return () => window.removeEventListener('keydown', handleKeyDown)
    77→  }, [isOpen, onClose, resetView])
    78→
    79→  const handleWheel = useCallback((e: React.WheelEvent) => {
    80→    e.preventDefault()
    81→    setZoom(z => {
    82→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    83→      if (next <= 1) setPosition({ x: 0, y: 0 })
    84→      return next
    85→    })
    86→  }, [])
    87→
    88→  const handleMouseDown = (e: React.MouseEvent) => {
    89→    if (zoom <= 1) return
    90→    setIsDragging(true)
    91→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    92→  }
    93→
    94→  const handleMouseMove = (e: React.MouseEvent) => {
    95→    if (!isDragging) return
    96→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
    97→  }
    98→
    99→  const handleMouseUp = () => setIsDragging(false)
   100→
   101→  const handleTouchStart = (e: React.TouchEvent) => {
   102→    if (e.touches.length === 2) {
   103→      const dx = e.touches[0].clientX - e.touches[1].clientX
   104→      const dy = e.touches[0].clientY - e.touches[1].clientY
   105→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   106→    } else if (e.touches.length === 1 && zoom > 1) {
   107→      setIsDragging(true)
   108→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   109→    }
   110→  }
   111→
   112→  const handleTouchMove = (e: React.TouchEvent) => {
   113→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   114→      const dx = e.touches[0].clientX - e.touches[1].clientX
   115→      const dy = e.touches[0].clientY - e.touches[1].clientY
   116→      const dist = Math.sqrt(dx * dx + dy * dy)
   117→      const scale = dist / lastTouchDist.current
   118→      setZoom(z => {
   119→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   120→        if (next <= 1) setPosition({ x: 0, y: 0 })
   121→        return next
   122→      })
   123→      lastTouchDist.current = dist
   124→    } else if (e.touches.length === 1 && isDragging) {
   125→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   126→    }
   127→  }
   128→
   129→  const handleTouchEnd = () => {
   130→    lastTouchDist.current = null
   131→    setIsDragging(false)
   132→  }
   133→
   134→  if (!isOpen) return null
   135→
   136→  return (
   137→    <div
   138→      className="fixed inset-0 z-[9999] bg-black/90 flex flex-col items-center justify-center"
   139→      onClick={(e) => e.stopPropagation()}
   140→      onMouseDown={(e) => e.stopPropagation()}
   141→    >
   142→      <div
   143→        ref={containerRef}
   144→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   145→        onWheel={handleWheel}
   146→        onMouseDown={handleMouseDown}
   147→        onMouseMove={handleMouseMove}
   148→        onMouseUp={handleMouseUp}
   149→        onMouseLeave={handleMouseUp}
   150→        onTouchStart={handleTouchStart}
   151→        onTouchMove={handleTouchMove}
   152→        onTouchEnd={handleTouchEnd}
   153→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   154→      >
   155→        {!imageLoaded && (
   156→          <Loader2 className="size-10 text-white animate-spin" />
   157→        )}
   158→        <img
   159→          src={imageUrl}
   160→          alt={label}
   161→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   162→          style={{
   163→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   164→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   165→            display: imageLoaded ? 'block' : 'none',
   166→          }}
   167→          onLoad={() => setImageLoaded(true)}
   168→          draggable={false}
   169→        />
   170→      </div>
   171→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   172→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   173→          <ZoomOut className="size-5" />
   174→        </Button>
   175→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   176→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   177→          <ZoomIn className="size-5" />
   178→        </Button>
   179→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   180→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   181→        </Button>
   182→        <span className="text-white/70 text-sm ml-2">{label}</span>
   183→      </div>
   184→    </div>
   185→  )
   186→}

   188→// ===== Exercise Detail Dialog =====
   189→function ExerciseDetailDialog({
   190→  exercise,
   191→  chapter,
   192→  exercises,
   193→  onClose,
   194→}: {
   195→  exercise: ExerciseData
   196→  chapter: ChapterData
   197→  exercises: ExerciseData[]
   198→  onClose: () => void
   199→}) {
   200→  const { user, setSelectedExercise } = useAppStore()
   201→  const student = user as StudentUser
   202→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   203→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   204→  const [saving, setSaving] = useState(false)
   205→  const [comments, setComments] = useState<CommentData[]>([])
   206→  const [newComment, setNewComment] = useState('')
   207→  const [sendingComment, setSendingComment] = useState(false)
   208→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   209→
   210→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   211→  const totalExercises = exercises.length
   212→
   213→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   214→
   215→  useEffect(() => {
   216→    let cancelled = false
   217→    if (commentsLoaded) return
   218→    const load = async () => {
   219→      try {
   220→        const res = await fetch('/api/comments')
   221→        if (res.ok && !cancelled) {
   222→          const data = await res.json()
   223→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   224→        }
   225→      } catch { /* ignore */ }
   226→      if (!cancelled) setCommentsLoaded(true)
   227→    }
   228→    load()
   229→  }, [exercise.id, commentsLoaded])
   230→
   231→  const refetchComments = async () => {
   232→    try {
   233→      const res = await fetch('/api/comments')
   234→      if (res.ok) {
   235→        const data = await res.json()
   236→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   237→      }
   238→    } catch { /* ignore */ }
   239→  }
   240→
   241→  const handleSave = async () => {
   242→    setSaving(true)
   243→    try {
   244→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   245→        method: 'PUT',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   248→      })
   249→      toast.success('Progression sauvegardée !')
   250→    } catch {
   251→      toast.error('Erreur lors de la sauvegarde.')
   252→    }
   253→    setSaving(false)
   254→  }
   255→
   256→  const handleSendComment = async () => {
   257→    if (!newComment.trim()) return
   258→    setSendingComment(true)
   259→    try {
   260→      await fetch('/api/comments', {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   264→      })
   265→      setNewComment('')
   266→      refetchComments()
   267→    } catch {
   268→      toast.error("Erreur lors de l'envoi.")
   269→    }
   270→    setSendingComment(false)
   271→  }
   272→
   273→  const goToExercise = (direction: 'prev' | 'next') => {
   274→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   275→    if (newIndex < 0 || newIndex >= totalExercises) return
   276→    setSelectedExercise(exercises[newIndex])
   277→  }
   278→
   279→  // Build list of page images for this exercise
   280→  const pageImages: { url: string; pageNum: number }[] = []
   281→  if (exercise.pageStart) {
   282→    const start = exercise.pageStart
   283→    const end = exercise.pageEnd || start
   284→    for (let p = start; p <= end; p++) {
   285→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   286→    }
   287→  }
   288→
   289→  return (
   290→    <>
   291→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   292→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   293→          <DialogHeader>
   294→            <DialogTitle className="text-lg">
   295→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   296→            </DialogTitle>
   297→            <DialogDescription className="sr-only">
   298→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   299→            </DialogDescription>
   300→          </DialogHeader>
   301→
   302→          {pageImages.length > 0 && (
   303→            <div className="space-y-3">
   304→              {pageImages.map((img) => (
   305→                <div key={img.pageNum} className="space-y-1.5">
   306→                  <div
   307→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   308→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   309→                  >
   310→                    <img
   311→                      src={img.url}
   312→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   313→                      className="w-full h-auto object-contain"
   314→                      loading="lazy"
   315→                    />
   316→                  </div>
   317→                  <Button
   318→                    variant="outline"
   319→                    size="sm"
   320→                    className="w-full"
   321→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   322→                  >
   323→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   324→                  </Button>
   325→                </div>
   326→              ))}
   327→            </div>
   328→          )}
   329→
   330→          <div className="space-y-2">
   331→            <Label className="text-sm font-medium">Statut</Label>
   332→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   333→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   334→                <RadioGroupItem value="not_started" />
   335→                <Circle className="size-4 text-gray-500" />
   336→                <span className="text-sm text-gray-500">Non commencé</span>
   337→              </label>
   338→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   339→                <RadioGroupItem value="in_progress" />
   340→                <Clock className="size-4 text-amber-600" />
   341→                <span className="text-sm text-amber-700">En cours</span>
   342→              </label>
   343→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   344→                <RadioGroupItem value="completed" />
   345→                <CheckCircle className="size-4 text-emerald-600" />
   346→                <span className="text-sm text-emerald-700">Terminé</span>
   347→              </label>
   348→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   349→                <RadioGroupItem value="difficulty" />
   350→                <AlertTriangle className="size-4 text-red-600" />
   351→                <span className="text-sm text-red-700">Difficulté</span>
   352→              </label>
   353→            </RadioGroup>
   354→          </div>
   355→
   356→          <div className="space-y-2">
   357→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   358→            <Textarea
   359→              id="notes"
   360→              value={note}
   361→              onChange={(e) => setNote(e.target.value)}
   362→              placeholder="Ajoutez vos notes ici..."
   363→              rows={3}
   364→            />
   365→          </div>
   366→
   367→          <div className="flex justify-end">
   368→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   369→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   370→              Sauvegarder
   371→            </Button>
   372→          </div>
   373→
   374→          <Separator />
   375→
   376→          <div className="space-y-3">
   377→            <h4 className="font-medium text-sm">Commentaires</h4>
   378→            {comments.map((comment) => (
   379→              <div key={comment.id} className="space-y-2">
   380→                <div className="flex items-start gap-2">
   381→                  <Avatar className="size-8">
   382→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   383→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   384→                    </AvatarFallback>
   385→                  </Avatar>
   386→                  <div className="flex-1">
   387→                    <div className="flex items-center gap-2">
   388→                      <span className="text-sm font-medium">
   389→                        {comment.student.firstName} {comment.student.lastName}
   390→                      </span>
   391→                      <span className="text-xs text-muted-foreground">
   392→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   393→                      </span>
   394→                    </div>
   395→                    <p className="text-sm mt-1">{comment.content}</p>
   396→                  </div>
   397→                </div>
   398→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   399→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   400→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   401→                    <span className="text-xs text-emerald-600 mt-1 block">
   402→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   403→                    </span>
   404→                  </div>
   405→                ))}
   406→              </div>
   407→            ))}
   408→            <div className="flex gap-2">
   409→              <Textarea
   410→                value={newComment}
   411→                onChange={(e) => setNewComment(e.target.value)}
   412→                placeholder="Ajouter un commentaire..."
   413→                rows={2}
   414→              />
   415→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   416→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   417→              </Button>
   418→            </div>
   419→          </div>
   420→
   421→          <DialogFooter className="flex-row justify-between">
   422→            <Button
   423→              variant="outline"
   424→              onClick={() => goToExercise('prev')}
   425→              disabled={currentIndex <= 0}
   426→            >
   427→              <ChevronLeft className="size-4 mr-1" /> Précédent
   428→            </Button>
   429→            <Button
   430→              variant="outline"
   431→              onClick={() => goToExercise('next')}
   432→              disabled={currentIndex >= totalExercises - 1}
   433→            >
   434→              Suivant <ChevronRight className="size-4 ml-1" />
   435→            </Button>
   436→          </DialogFooter>
   437→
   438→          <ImageLightbox
   439→            key={lightbox.imageUrl}
   440→            isOpen={lightbox.isOpen}
   441→            imageUrl={lightbox.imageUrl}
   442→            label={lightbox.label}
   443→            onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   444→          />
   445→        </DialogContent>
   446→      </Dialog>
   447→    </>
   448→  )
   449→}

   462→  // Register form

   464→  const [regLastName, setRegLastName] = useState('')
   465→  const [regEmail, setRegEmail] = useState('')
   466→  const [regClassName, setRegClassName] = useState('')
   467→  const [regSchoolName, setRegSchoolName] = useState('')
   468→  const [regPhone, setRegPhone] = useState('')

   473→
   474→  // Form states are initialized to defaults to prevent autofill
   475→
   476→  const handleLogin = async () => {

   479→      const res = await fetch('/api/student/login', {
   480→        method: 'POST',

   488→      } else {
   489→        const data = await res.json().catch(() => ({}))
   490→        toast.error(data.error || 'Identifiants incorrects.')
   491→      }
   492→    } catch {
   493→      toast.error('Erreur de connexion.')
   494→    }
   495→    setLoading(false)
   496→  }
   497→
   498→  const handleRegister = async () => {
   499→    setLoading(true)
   500→    try {
   501→      const res = await fetch('/api/student/register', {
   502→        method: 'POST',
   503→        headers: { 'Content-Type': 'application/json' },
   504→        body: JSON.stringify({
   505→          firstName: regFirstName, lastName: regLastName, email: regEmail,
   506→          className: regClassName, schoolName: regSchoolName, phone: regPhone, password: regPassword,

   776→                      onImageClick={(ex) => {
   777→                        if (ex.pageStart) {
   778→                          setLightbox({ isOpen: true, imageUrl: pageImageUrl(ex.pageStart!), label: `Ex.${ex.number}` })
   779→                        }
   780→                      }}
   781→                    />
   782→                  ))}

   784→              </AccordionContent>

   847→  isLoading: boolean

   852→}) {
