    60→  const containerRef = useRef<HTMLDivElement>(null)
    61→  const lastTouchDist = useRef<number | null>(null)
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) {
    70→      resetView()
    71→      setImageLoaded(false)
    72→    }
    73→  }, [isOpen, resetView])
    74→
    75→  useEffect(() => {
    76→    if (!isOpen) return
    77→    const handleKeyDown = (e: KeyboardEvent) => {
    78→      if (e.key === 'Escape') onClose()
    79→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    80→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    81→      else if (e.key === '0') resetView()
    82→    }
    83→    window.addEventListener('keydown', handleKeyDown)
    84→    return () => window.removeEventListener('keydown', handleKeyDown)
    85→  }, [isOpen, onClose, resetView])
    86→
    87→  const handleWheel = useCallback((e: React.WheelEvent) => {
    88→    e.preventDefault()
    89→    setZoom(z => {
    90→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    91→      if (next <= 1) setPosition({ x: 0, y: 0 })
    92→      return next
    93→    })
    94→  }, [])
    95→
    96→  const handleMouseDown = (e: React.MouseEvent) => {
    97→    if (zoom <= 1) return
    98→    setIsDragging(true)
    99→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
   100→  }
   101→
   102→  const handleMouseMove = (e: React.MouseEvent) => {
   103→    if (!isDragging) return
   104→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
   105→  }
   106→
   107→  const handleMouseUp = () => setIsDragging(false)
   108→
   109→  const handleTouchStart = (e: React.TouchEvent) => {
   110→    if (e.touches.length === 2) {
   111→      const dx = e.touches[0].clientX - e.touches[1].clientX
   112→      const dy = e.touches[0].clientY - e.touches[1].clientY
   113→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   114→    } else if (e.touches.length === 1 && zoom > 1) {
   115→      setIsDragging(true)
   116→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   117→    }
   118→  }
   119→
   120→  const handleTouchMove = (e: React.TouchEvent) => {
   121→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   122→      const dx = e.touches[0].clientX - e.touches[1].clientX
   123→      const dy = e.touches[0].clientY - e.touches[1].clientY
   124→      const dist = Math.sqrt(dx * dx + dy * dy)
   125→      const scale = dist / lastTouchDist.current
   126→      setZoom(z => {
   127→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   128→        if (next <= 1) setPosition({ x: 0, y: 0 })
   129→        return next
   130→      })
   131→      lastTouchDist.current = dist
   132→    } else if (e.touches.length === 1 && isDragging) {
   133→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   134→    }
   135→  }
   136→
   137→  const handleTouchEnd = () => {
   138→    lastTouchDist.current = null
   139→    setIsDragging(false)
   140→  }
   141→
   142→  if (!isOpen) return null
   143→
   144→  return createPortal(
   145→    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col items-center justify-center">
   146→      <div
   147→        ref={containerRef}
   148→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   149→        onWheel={handleWheel}
   150→        onMouseDown={handleMouseDown}
   151→        onMouseMove={handleMouseMove}
   152→        onMouseUp={handleMouseUp}
   153→        onMouseLeave={handleMouseUp}
   154→        onTouchStart={handleTouchStart}
   155→        onTouchMove={handleTouchMove}
   156→        onTouchEnd={handleTouchEnd}
   157→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   158→      >
   159→        {!imageLoaded && (
   160→          <Loader2 className="size-10 text-white animate-spin" />
   161→        )}
   162→        <img
   163→          src={imageUrl}
   164→          alt={label}
   165→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   166→          style={{
   167→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   168→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   169→            display: imageLoaded ? 'block' : 'none',
   170→          }}
   171→          onLoad={() => setImageLoaded(true)}
   172→          draggable={false}
   173→        />
   174→      </div>
   175→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   176→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   177→          <ZoomOut className="size-5" />
   178→        </Button>
   179→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   180→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   181→          <ZoomIn className="size-5" />
   182→        </Button>
   183→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   184→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   185→        </Button>
   186→        <span className="text-white/70 text-sm ml-2">{label}</span>
   187→      </div>
   188→    </div>,
   189→    document.body
   190→  )
   191→}
