    37→    not_started: { label: 'Non commencé', className: 'bg-gray-100 text-gray-500' },

    46→// ===== Image Lightbox (Portal) =====
    47→function ImageLightbox({
    48→  isOpen, imageUrl, label, onClose
    49→}: {
    50→  isOpen: boolean
    51→  imageUrl: string
    52→  label: string
    53→  onClose: () => void
    54→}) {
    55→  const [zoom, setZoom] = useState(1)
    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→  const [isDragging, setIsDragging] = useState(false)
    58→  const [imageLoaded, setImageLoaded] = useState(false)
    59→  const containerRef = useRef<HTMLDivElement>(null)
    60→  const lastTouchDist = useRef<number | null>(null)
    61→  const dragStartRef = useRef({ x: 0, y: 0 })
    62→  const posStartRef = useRef({ x: 0, y: 0 })
    63→
    64→  const resetView = useCallback(() => {
    65→    setZoom(1)
    66→    setPosition({ x: 0, y: 0 })
    67→  }, [])
    68→
    69→  useEffect(() => {
    70→    if (!isOpen) return
    71→    const handleKeyDown = (e: KeyboardEvent) => {
    72→      if (e.key === 'Escape') {
    73→        e.stopPropagation()
    74→        e.preventDefault()
    75→        onClose()
    76→      }
    77→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    78→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    79→      else if (e.key === '0') resetView()
    80→    }
    81→    window.addEventListener('keydown', handleKeyDown, true)
    82→    return () => window.removeEventListener('keydown', handleKeyDown, true)
    83→  }, [isOpen, onClose, resetView])
    84→
    85→  // Global mousemove/mouseup for smooth drag even when cursor leaves container
    86→  useEffect(() => {
    87→    if (!isDragging) return
    88→    const handleGlobalMove = (e: MouseEvent) => {
    89→      setPosition({
    90→        x: posStartRef.current.x + (e.clientX - dragStartRef.current.x),
    91→        y: posStartRef.current.y + (e.clientY - dragStartRef.current.y),
    92→      })
    93→    }
    94→    const handleGlobalUp = () => {
    95→      setIsDragging(false)
    96→    }
    97→    window.addEventListener('mousemove', handleGlobalMove)
    98→    window.addEventListener('mouseup', handleGlobalUp)
    99→    return () => {
   100→      window.removeEventListener('mousemove', handleGlobalMove)
   101→      window.removeEventListener('mouseup', handleGlobalUp)
   102→    }
   103→  }, [isDragging])
   104→
   105→  const handleWheel = useCallback((e: React.WheelEvent) => {
   106→    e.preventDefault()
   107→    setZoom(z => {
   108→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
   109→      if (next <= 1) setPosition({ x: 0, y: 0 })
   110→      return next
   111→    })
   112→  }, [])
   113→
   114→  const handlePointerDown = (e: React.PointerEvent) => {
   115→    if (zoom <= 1) return
   116→    e.preventDefault()
   117→    setIsDragging(true)
   118→    dragStartRef.current = { x: e.clientX, y: e.clientY }
   119→    posStartRef.current = { x: position.x, y: position.y }
   120→  }
   121→
   122→  const handleTouchStart = (e: React.TouchEvent) => {
   123→    if (e.touches.length === 2) {
   124→      const dx = e.touches[0].clientX - e.touches[1].clientX
   125→      const dy = e.touches[0].clientY - e.touches[1].clientY
   126→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   127→    } else if (e.touches.length === 1 && zoom > 1) {
   128→      setIsDragging(true)
   129→      dragStartRef.current = { x: e.touches[0].clientX, y: e.touches[0].clientY }
   130→      posStartRef.current = { x: position.x, y: position.y }
   131→    }
   132→  }
   133→
   134→  const handleTouchMove = (e: React.TouchEvent) => {
   135→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   136→      const dx = e.touches[0].clientX - e.touches[1].clientX
   137→      const dy = e.touches[0].clientY - e.touches[1].clientY
   138→      const dist = Math.sqrt(dx * dx + dy * dy)
   139→      const scale = dist / lastTouchDist.current
   140→      setZoom(z => {
   141→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   142→        if (next <= 1) setPosition({ x: 0, y: 0 })
   143→        return next
   144→      })
   145→      lastTouchDist.current = dist
   146→    } else if (e.touches.length === 1 && isDragging) {
   147→      setPosition({
   148→        x: posStartRef.current.x + (e.touches[0].clientX - dragStartRef.current.x),
   149→        y: posStartRef.current.y + (e.touches[0].clientY - dragStartRef.current.y),
   150→      })
   151→    }
   152→  }
   153→
   154→  const handleTouchEnd = () => {
   155→    lastTouchDist.current = null
   156→    setIsDragging(false)
   157→  }
   158→
   159→
   160→  // Close lightbox when clicking the dark overlay area (not the image or controls)
   161→  const handleOverlayClick = useCallback((e: React.MouseEvent) => {
   162→    // Only close if clicking directly on the overlay, not on children
   163→    if (e.target === e.currentTarget) {
   164→      onClose()
   165→    }
   166→  }, [onClose])
   167→
   168→  if (!isOpen) return null
   169→
   170→  const lightboxContent = (
   171→    <div
   172→      className="fixed inset-0 z-[9999] bg-black/90 flex flex-col items-center justify-center pointer-events-auto"
   173→      onClick={handleOverlayClick}
   174→    >
   175→      {/* Close button */}
   176→      <button
   177→        onClick={(e) => { e.stopPropagation(); onClose() }}
   178→        className="absolute top-4 right-4 z-10 text-white/70 hover:text-white transition-colors p-2 rounded-full hover:bg-white/10"
   179→        aria-label="Fermer"
   180→      >
   181→        <X className="size-6" />
   182→      </button>
   183→      <div
   184→        ref={containerRef}
   185→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   186→        onWheel={handleWheel}
   187→        onPointerDown={handlePointerDown}
   188→        onTouchStart={handleTouchStart}
   189→        onTouchMove={handleTouchMove}
   190→        onTouchEnd={handleTouchEnd}
   191→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default', userSelect: 'none' }}
   192→      >
   193→        {!imageLoaded && (
   194→          <Loader2 className="size-10 text-white animate-spin" />
   195→        )}
   196→        <img
   197→          src={imageUrl}
   198→          alt={label}
   199→          className="max-w-[90vw] max-h-[80vh] object-contain select-none pointer-events-none"
   200→          style={{
   201→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   202→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   203→            display: imageLoaded ? 'block' : 'none',
   204→          }}
   205→          onLoad={() => setImageLoaded(true)}
   206→          draggable={false}
   207→        />
   208→      </div>
   209→      {/* Zoom control bar */}
   210→      <div
   211→        className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl"
   212→        onClick={(e) => e.stopPropagation()}
   213→      >
   214→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.max(z - 0.25, 0.5)) }}>
   215→          <ZoomOut className="size-5" />
   216→        </Button>
   217→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   218→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.min(z + 0.25, 5)) }}>
   219→          <ZoomIn className="size-5" />
   220→        </Button>
   221→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); resetView() }}>
   222→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   223→        </Button>
   224→        <span className="text-white/70 text-sm ml-2">{label}</span>
   225→      </div>
   226→    </div>
   227→  )
   228→
   229→  // Render via portal to document.body so it sits at the same DOM level as the
   230→  // Radix Dialog portal and is guaranteed to be on top (z-9999 > dialog z-50).
   231→  if (typeof document === 'undefined') return null
   232→  return createPortal(lightboxContent, document.body)
   233→}

   235→// ===== Exercise Detail Dialog =====
   236→function ExerciseDetailDialog({
   237→  exercise,
   238→  chapter,
   239→  exercises,
   240→  onClose,
   241→}: {
   242→  exercise: ExerciseData
   243→  chapter: ChapterData
   244→  exercises: ExerciseData[]
   245→  onClose: () => void
   246→}) {
   247→  const { user, setSelectedExercise, openLightbox, lightbox } = useAppStore()
   248→  const student = user as StudentUser
   249→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   250→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   251→  const [saving, setSaving] = useState(false)
   252→  const [comments, setComments] = useState<CommentData[]>([])
   253→  const [newComment, setNewComment] = useState('')
   254→  const [sendingComment, setSendingComment] = useState(false)
   255→
   256→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   257→  const totalExercises = exercises.length
   258→
   259→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   260→
   261→  useEffect(() => {
   262→    let cancelled = false
   263→    if (commentsLoaded) return
   264→    const load = async () => {
   265→      try {
   266→        const res = await fetch('/api/comments')
   267→        if (res.ok && !cancelled) {
   268→          const data = await res.json()
   269→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   270→        }
   271→      } catch { /* ignore */ }
   272→      if (!cancelled) setCommentsLoaded(true)
   273→    }
   274→    load()
   275→  }, [exercise.id, commentsLoaded])
   276→
   277→  const refetchComments = async () => {
   278→    try {
   279→      const res = await fetch('/api/comments')
   280→      if (res.ok) {
   281→        const data = await res.json()
   282→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   283→      }
   284→    } catch { /* ignore */ }
   285→  }
   286→
   287→  const handleSave = async () => {
   288→    setSaving(true)
   289→    try {
   290→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   291→        method: 'PUT',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   294→      })
   295→      toast.success('Progression sauvegardée !')
   296→    } catch {
   297→      toast.error('Erreur lors de la sauvegarde.')
   298→    }
   299→    setSaving(false)
   300→  }
   301→
   302→  const handleSendComment = async () => {
   303→    if (!newComment.trim()) return
   304→    setSendingComment(true)
   305→    try {
   306→      await fetch('/api/comments', {
   307→        method: 'POST',
   308→        headers: { 'Content-Type': 'application/json' },
   309→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   310→      })
   311→      setNewComment('')
   312→      refetchComments()
   313→    } catch {
   314→      toast.error("Erreur lors de l'envoi.")
   315→    }
   316→    setSendingComment(false)
   317→  }
   318→
   319→  const goToExercise = (direction: 'prev' | 'next') => {
   320→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   321→    if (newIndex < 0 || newIndex >= totalExercises) return
   322→    setSelectedExercise(exercises[newIndex])
   323→  }
   324→
   325→  // Build list of page images for this exercise
   326→  const pageImages: { url: string; pageNum: number }[] = []
   327→  if (exercise.pageStart) {
   328→    const start = exercise.pageStart
   329→    const end = exercise.pageEnd || start
   330→    for (let p = start; p <= end; p++) {
   331→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   332→    }
   333→  }
   334→
   335→  return (
   336→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   337→        <DialogContent 
   338→          className="max-w-4xl max-h-[90vh] overflow-y-auto"
   339→          onInteractOutside={(e) => { if (lightbox.isOpen) e.preventDefault() }}
   340→        >
   341→          <DialogHeader>
   342→            <DialogTitle className="text-lg">
   343→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   344→            </DialogTitle>
   345→            <DialogDescription className="sr-only">
   346→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   347→            </DialogDescription>
   348→          </DialogHeader>
   349→
   350→          {pageImages.length > 0 && (
   351→            <div className="space-y-3">
   352→              {pageImages.map((img) => (
   353→                <div key={img.pageNum} className="space-y-1.5">
   354→                  <div
   355→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   356→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   357→                  >
   358→                    <img
   359→                      src={img.url}
   360→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   361→                      className="w-full h-auto object-contain"
   362→                      loading="lazy"
   363→                    />
   364→                  </div>
   365→                  <Button
   366→                    variant="outline"
   367→                    size="sm"
   368→                    className="w-full"
   369→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   370→                  >
   371→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   372→                  </Button>
   373→                </div>
   374→              ))}
   375→            </div>
   376→          )}
   377→
   378→          <div className="space-y-2">
   379→            <Label className="text-sm font-medium">Statut</Label>
   380→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   381→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   382→                <RadioGroupItem value="not_started" />
   383→                <Circle className="size-4 text-gray-500" />
   384→                <span className="text-sm text-gray-500">Non commencé</span>
   385→              </label>
   386→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   387→                <RadioGroupItem value="in_progress" />
   388→                <Clock className="size-4 text-amber-600" />
   389→                <span className="text-sm text-amber-700">En cours</span>
   390→              </label>
   391→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   392→                <RadioGroupItem value="completed" />
   393→                <CheckCircle className="size-4 text-emerald-600" />
   394→                <span className="text-sm text-emerald-700">Terminé</span>
   395→              </label>
   396→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   397→                <RadioGroupItem value="difficulty" />
   398→                <AlertTriangle className="size-4 text-red-600" />
   399→                <span className="text-sm text-red-700">Difficulté</span>
   400→              </label>
   401→            </RadioGroup>
   402→          </div>
   403→
   404→          <div className="space-y-2">
   405→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   406→            <Textarea
   407→              id="notes"
   408→              value={note}
   409→              onChange={(e) => setNote(e.target.value)}
   410→              placeholder="Ajoutez vos notes ici..."
   411→              rows={3}
   412→            />
   413→          </div>
   414→
   415→          <div className="flex justify-end">
   416→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   417→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   418→              Sauvegarder
   419→            </Button>
   420→          </div>
   421→
   422→          <Separator />
   423→
   424→          <div className="space-y-3">
   425→            <h4 className="font-medium text-sm">Commentaires</h4>
   426→            {comments.map((comment) => (
   427→              <div key={comment.id} className="space-y-2">
   428→                <div className="flex items-start gap-2">
   429→                  <Avatar className="size-8">
   430→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   431→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   432→                    </AvatarFallback>
   433→                  </Avatar>
   434→                  <div className="flex-1">
   435→                    <div className="flex items-center gap-2">
   436→                      <span className="text-sm font-medium">
   437→                        {comment.student.firstName} {comment.student.lastName}
   438→                      </span>
   439→                      <span className="text-xs text-muted-foreground">
   440→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   441→                      </span>
   442→                    </div>
   443→                    <p className="text-sm mt-1">{comment.content}</p>
   444→                  </div>
   445→                </div>
   446→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   447→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   448→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   449→                    <span className="text-xs text-emerald-600 mt-1 block">
   450→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   451→                    </span>
   452→                  </div>
   453→                ))}
   454→              </div>
   455→            ))}
   456→            <div className="flex gap-2">
   457→              <Textarea
   458→                value={newComment}
   459→                onChange={(e) => setNewComment(e.target.value)}
   460→                placeholder="Ajouter un commentaire..."
   461→                rows={2}
   462→              />
   463→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   464→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   465→              </Button>
   466→            </div>
   467→          </div>
   468→
   469→          <DialogFooter className="flex-row justify-between">
   470→            <Button
   471→              variant="outline"
   472→              onClick={() => goToExercise('prev')}
   473→              disabled={currentIndex <= 0}
   474→            >
   475→              <ChevronLeft className="size-4 mr-1" /> Précédent
   476→            </Button>
   477→            <Button
   478→              variant="outline"
   479→              onClick={() => goToExercise('next')}
   480→              disabled={currentIndex >= totalExercises - 1}
   481→            >
   482→              Suivant <ChevronRight className="size-4 ml-1" />
   483→            </Button>
   484→          </DialogFooter>
   485→
   486→        </DialogContent>
   487→      </Dialog>
   488→  )
   489→}

   749→    let cancelled = false

   751→      try {

   770→    setLoadingExercises(chapter.id)
   771→    try {
   772→      const res = await fetch(`/api/chapters/${chapter.id}/exercises?studentId=${student.id}`)
   773→      if (res.ok) {
   774→        const data = await res.json()
   775→        setChapterExercises(prev => ({ ...prev, [chapter.id]: data.exercises || [] }))

   808→                      key={ch.id}
   809→                      chapter={ch}
   810→                      isExpanded={expandedChapterId === ch.id}
   811→                      isLoading={loadingExercises === ch.id}
   812→                      exercises={chapterExercises[ch.id] || []}
   813→                      onToggle={() => toggleChapter(ch)}
   814→                      onExerciseClick={(ex) => setSelectedExercise(ex)}
   815→                      onImageClick={(ex) => {
   816→                        if (ex.pageStart) {
   817→                          openLightbox(pageImageUrl(ex.pageStart!), `Ex.${ex.number}`)
   818→                        }
   819→                      }}
   820→                    />
   821→                  ))}
   822→                </div>
   823→              </AccordionContent>
   824→            </AccordionItem>
   825→          )}
   826→
   827→          {algebre.length > 0 && (

   842→                        if (ex.pageStart) {

   875→function ChapterRowInline({ chapter, isExpanded, isLoading, exercises, onToggle, onExerciseClick, onImageClick }: {
   876→  chapter: ChapterData
   877→  isExpanded: boolean
   878→  isLoading: boolean
   879→  exercises: ExerciseData[]
   880→  onToggle: () => void
   881→  onExerciseClick: (ex: ExerciseData) => void
   882→  onImageClick: (ex: ExerciseData) => void
   883→}) {
   884→  const allDone = chapter.completedCount === chapter.exerciseCount
   885→  return (
   886→    <div className="rounded-lg border border-l-4 border-l-emerald-400 overflow-hidden">

   888→        onClick={onToggle}
   889→        className="w-full flex items-center gap-3 p-3 hover:bg-gray-50 transition-colors text-left"

   891→        <ChevronRight className={`size-4 text-muted-foreground transition-transform duration-200 ${isExpanded ? 'rotate-90' : ''}`} />
   892→        <span className="font-medium text-sm">Ch.{chapter.number}:</span>
   893→        <span className="flex-1 text-sm truncate">{chapter.title}</span>
   894→        <Badge variant="secondary" className={allDone ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'}>
   895→          {chapter.completedCount}/{chapter.exerciseCount}
   896→        </Badge>

   900→        <div className="border-t px-3 pb-3 pt-2">

  1553→function TeacherProfileView() {

  1556→  const [profile, setProfile] = useState<TeacherUser | null>(null)

  1560→  const [saving, setSaving] = useState(false)

  1562→
  1563→  useEffect(() => {
  1564→    const fetchProfile = async () => {
  1565→      try {
  1566→        const res = await fetch('/api/teacher/profile')
  1567→        if (res.ok) {
  1568→          const data = await res.json()
  1569→          setProfile(data)
  1570→        }
  1571→      } catch { /* ignore */ }
  1572→    }
  1573→    fetchProfile()
  1574→  }, [])
  1575→
  1576→  const displayTeacher = profile || teacher
  1577→
  1578→  const handleChangePassword = async () => {
  1579→    setError('')

  1581→      setError('Tous les champs sont requis.')

  1589→      setError('Les mots de passe ne correspondent pas.')
  1590→      return
  1591→    }
  1592→    setSaving(true)

  1594→      const res = await fetch('/api/teacher/profile', {
  1595→        method: 'PUT',
  1596→        headers: { 'Content-Type': 'application/json' },
  1597→        body: JSON.stringify({ currentPassword, newPassword }),
  1598→      })
  1599→      if (res.ok) {

  1601→        setCurrentPassword('')
  1602→        setNewPassword('')
  1603→        setConfirmPassword('')
  1604→      } else {
  1605→        setError('Mot de passe actuel incorrect.')
  1606→      }
  1607→    } catch {
  1608→      setError('Erreur lors de la modification.')
  1609→    }
  1610→    setSaving(false)
  1611→  }

  1613→  return (
