    46→// ===== Image Lightbox (Portal) =====
    47→function ImageLightbox({
    48→  isOpen, imageUrl, label, onClose
    49→}: {
    50→  isOpen: boolean
    51→  imageUrl: string
    52→  label: string
    53→  onClose: () => void
    54→}) {
    55→  const [zoom, setZoom] = useState(1)
    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→  const [isDragging, setIsDragging] = useState(false)
    58→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    59→  const [imageLoaded, setImageLoaded] = useState(false)
    60→  const containerRef = useRef<HTMLDivElement>(null)
    61→  const lastTouchDist = useRef<number | null>(null)
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) return
    70→    const handleKeyDown = (e: KeyboardEvent) => {
    71→      if (e.key === 'Escape') onClose()
    72→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    73→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    74→      else if (e.key === '0') resetView()
    75→    }
    76→    window.addEventListener('keydown', handleKeyDown)
    77→    return () => window.removeEventListener('keydown', handleKeyDown)
    78→  }, [isOpen, onClose, resetView])
    79→
    80→  const handleWheel = useCallback((e: React.WheelEvent) => {
    81→    e.preventDefault()
    82→    setZoom(z => {
    83→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    84→      if (next <= 1) setPosition({ x: 0, y: 0 })
    85→      return next
    86→    })
    87→  }, [])
    88→
    89→  const handleMouseDown = (e: React.MouseEvent) => {
    90→    if (zoom <= 1) return
    91→    setIsDragging(true)
    92→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    93→  }
    94→
    95→  const handleMouseMove = (e: React.MouseEvent) => {
    96→    if (!isDragging) return
    97→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
    98→  }
    99→
   100→  const handleMouseUp = () => setIsDragging(false)
   101→
   102→  const handleTouchStart = (e: React.TouchEvent) => {
   103→    if (e.touches.length === 2) {
   104→      const dx = e.touches[0].clientX - e.touches[1].clientX
   105→      const dy = e.touches[0].clientY - e.touches[1].clientY
   106→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   107→    } else if (e.touches.length === 1 && zoom > 1) {
   108→      setIsDragging(true)
   109→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   110→    }
   111→  }
   112→
   113→  const handleTouchMove = (e: React.TouchEvent) => {
   114→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   115→      const dx = e.touches[0].clientX - e.touches[1].clientX
   116→      const dy = e.touches[0].clientY - e.touches[1].clientY
   117→      const dist = Math.sqrt(dx * dx + dy * dy)
   118→      const scale = dist / lastTouchDist.current
   119→      setZoom(z => {
   120→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   121→        if (next <= 1) setPosition({ x: 0, y: 0 })
   122→        return next
   123→      })
   124→      lastTouchDist.current = dist
   125→    } else if (e.touches.length === 1 && isDragging) {
   126→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   127→    }
   128→  }
   129→
   130→  const handleTouchEnd = () => {
   131→    lastTouchDist.current = null
   132→    setIsDragging(false)
   133→  }
   134→
   135→  if (!isOpen) return null
   136→
   137→  return createPortal(
   138→    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col items-center justify-center">
   139→      <div
   140→        ref={containerRef}
   141→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   142→        onWheel={handleWheel}
   143→        onMouseDown={handleMouseDown}
   144→        onMouseMove={handleMouseMove}
   145→        onMouseUp={handleMouseUp}
   146→        onMouseLeave={handleMouseUp}
   147→        onTouchStart={handleTouchStart}
   148→        onTouchMove={handleTouchMove}
   149→        onTouchEnd={handleTouchEnd}
   150→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   151→      >
   152→        {!imageLoaded && (
   153→          <Loader2 className="size-10 text-white animate-spin" />
   154→        )}
   155→        <img
   156→          src={imageUrl}
   157→          alt={label}
   158→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   159→          style={{
   160→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   161→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   162→            display: imageLoaded ? 'block' : 'none',
   163→          }}
   164→          onLoad={() => setImageLoaded(true)}
   165→          draggable={false}
   166→        />
   167→      </div>
   168→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   169→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   170→          <ZoomOut className="size-5" />
   171→        </Button>
   172→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   173→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   174→          <ZoomIn className="size-5" />
   175→        </Button>
   176→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   177→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   178→        </Button>
   179→        <span className="text-white/70 text-sm ml-2">{label}</span>
   180→      </div>
   181→    </div>,
   182→    document.body
   183→  )
   184→}

   186→// ===== Exercise Detail Dialog =====
   187→function ExerciseDetailDialog({
   188→  exercise,
   189→  chapter,
   190→  onClose,
   191→}: {
   192→  exercise: ExerciseData
   193→  chapter: ChapterData
   194→  onClose: () => void
   195→}) {
   196→  const { user, chapterExercises, setSelectedExercise, setChapterExercises } = useAppStore()
   197→  const student = user as StudentUser
   198→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   199→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   200→  const [saving, setSaving] = useState(false)
   201→  const [comments, setComments] = useState<CommentData[]>([])
   202→  const [newComment, setNewComment] = useState('')
   203→  const [sendingComment, setSendingComment] = useState(false)
   204→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   205→
   206→  const currentIndex = chapterExercises.findIndex(e => e.id === exercise.id)
   207→  const totalExercises = chapterExercises.length
   208→
   209→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   210→
   211→  useEffect(() => {
   212→    let cancelled = false
   213→    const load = async () => {
   214→      try {
   215→        const res = await fetch('/api/comments')
   216→        if (res.ok && !cancelled) {
   217→          const data = await res.json()
   218→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   219→        }
   220→      } catch { /* ignore */ }
   221→      if (!cancelled) setCommentsLoaded(true)
   222→    }
   223→    if (!commentsLoaded) load()
   224→  }, [exercise.id, commentsLoaded])
   225→
   226→  const handleSave = async () => {
   227→    setSaving(true)
   228→    try {
   229→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   230→        method: 'PUT',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   233→      })
   234→      toast.success('Progression sauvegardée !')
   235→      const updatedExercises = chapterExercises.map(e =>
   236→        e.id === exercise.id ? { ...e, progress: { id: e.progress?.id || '', status, studentNote: note } } : e
   237→      )
   238→      setChapterExercises(updatedExercises)
   239→    } catch {
   240→      toast.error('Erreur lors de la sauvegarde.')
   241→    }
   242→    setSaving(false)
   243→  }
   244→
   245→  const handleSendComment = async () => {
   246→    if (!newComment.trim()) return
   247→    setSendingComment(true)
   248→    try {
   249→      await fetch('/api/comments', {
   250→        method: 'POST',
   251→        headers: { 'Content-Type': 'application/json' },
   252→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   253→      })
   254→      setNewComment('')
   255→      fetchComments()
   256→    } catch {
   257→      toast.error("Erreur lors de l'envoi.")
   258→    }
   259→    setSendingComment(false)
   260→  }
   261→
   262→  const goToExercise = (direction: 'prev' | 'next') => {
   263→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   264→    if (newIndex < 0 || newIndex >= totalExercises) return
   265→    setSelectedExercise(chapterExercises[newIndex])
   266→  }
   267→
   268→  const imageUrl = exercise.pageStart ? exerciseImageUrl(chapter.number, exercise.number) : ''
   269→
   270→  return (
   271→    <>
   272→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   273→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   274→          <DialogHeader>
   275→            <DialogTitle className="text-lg">
   276→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   277→            </DialogTitle>
   278→            <DialogDescription className="sr-only">
   279→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   280→            </DialogDescription>
   281→          </DialogHeader>
   282→
   283→          {imageUrl && (
   284→            <div className="space-y-2">
   285→              <div className="relative rounded-lg overflow-hidden border bg-gray-50 max-h-64">
   286→                <img src={imageUrl} alt={`Exercice ${exercise.number}`} className="w-full h-full object-contain max-h-64" />
   287→              </div>
   288→              <Button variant="outline" size="sm" onClick={() => setLightbox({ isOpen: true, imageUrl, label: `Ex.${exercise.number}` })}>
   289→                <Eye className="size-4 mr-1" /> Agrandir
   290→              </Button>
   291→            </div>
   292→          )}
   293→
   294→          <div className="space-y-2">
   295→            <Label className="text-sm font-medium">Statut</Label>
   296→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   297→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   298→                <RadioGroupItem value="not_started" />
   299→                <Circle className="size-4 text-gray-500" />
   300→                <span className="text-sm text-gray-500">Non commencé</span>
   301→              </label>
   302→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   303→                <RadioGroupItem value="in_progress" />
   304→                <Clock className="size-4 text-amber-600" />
   305→                <span className="text-sm text-amber-700">En cours</span>
   306→              </label>
   307→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   308→                <RadioGroupItem value="completed" />
   309→                <CheckCircle className="size-4 text-emerald-600" />
   310→                <span className="text-sm text-emerald-700">Terminé</span>
   311→              </label>
   312→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   313→                <RadioGroupItem value="difficulty" />
   314→                <AlertTriangle className="size-4 text-red-600" />
   315→                <span className="text-sm text-red-700">Difficulté</span>
   316→              </label>
   317→            </RadioGroup>
   318→          </div>
   319→
   320→          <div className="space-y-2">
   321→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   322→            <Textarea
   323→              id="notes"
   324→              value={note}
   325→              onChange={(e) => setNote(e.target.value)}
   326→              placeholder="Ajoutez vos notes ici..."
   327→              rows={3}
   328→            />
   329→          </div>
   330→
   331→          <div className="flex justify-end">
   332→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   333→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   334→              Sauvegarder
   335→            </Button>
   336→          </div>
   337→
   338→          <Separator />
   339→
   340→          <div className="space-y-3">
   341→            <h4 className="font-medium text-sm">Commentaires</h4>
   342→            {comments.map((comment) => (
   343→              <div key={comment.id} className="space-y-2">
   344→                <div className="flex items-start gap-2">
   345→                  <Avatar className="size-8">
   346→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   347→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   348→                    </AvatarFallback>
   349→                  </Avatar>
   350→                  <div className="flex-1">
   351→                    <div className="flex items-center gap-2">
   352→                      <span className="text-sm font-medium">
   353→                        {comment.student.firstName} {comment.student.lastName}
   354→                      </span>
   355→                      <span className="text-xs text-muted-foreground">
   356→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   357→                      </span>
   358→                    </div>
   359→                    <p className="text-sm mt-1">{comment.content}</p>
   360→                  </div>
   361→                </div>
   362→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   363→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   364→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   365→                    <span className="text-xs text-emerald-600 mt-1 block">
   366→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   367→                    </span>
   368→                  </div>
   369→                ))}
   370→              </div>
   371→            ))}
   372→            <div className="flex gap-2">
   373→              <Textarea
   374→                value={newComment}
   375→                onChange={(e) => setNewComment(e.target.value)}
   376→                placeholder="Ajouter un commentaire..."
   377→                rows={2}
   378→              />
   379→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   380→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   381→              </Button>
   382→            </div>
   383→          </div>
   384→
   385→          <DialogFooter className="flex-row justify-between">
   386→            <Button
   387→              variant="outline"
   388→              onClick={() => goToExercise('prev')}
   389→              disabled={currentIndex <= 0}
   390→            >
   391→              <ChevronLeft className="size-4 mr-1" /> Précédent
   392→            </Button>
   393→            <Button
   394→              variant="outline"
   395→              onClick={() => goToExercise('next')}
   396→              disabled={currentIndex >= totalExercises - 1}
   397→            >
   398→              Suivant <ChevronRight className="size-4 ml-1" />
   399→            </Button>
   400→          </DialogFooter>
   401→        </DialogContent>
   402→      </Dialog>
   403→
   404→      <ImageLightbox
   405→        key={lightbox.imageUrl}
   406→        isOpen={lightbox.isOpen}
   407→        imageUrl={lightbox.imageUrl}
   408→        label={lightbox.label}
   409→        onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   410→      />
   411→    </>
   412→  )
   413→}

   497→        toast.error('Identifiants incorrects.')
