    46→// ===== Image Lightbox (Portal) =====
    47→function ImageLightbox({
    48→  isOpen, imageUrl, label, onClose
    49→}: {
    50→  isOpen: boolean
    51→  imageUrl: string
    52→  label: string
    53→  onClose: () => void
    54→}) {
    55→  const [zoom, setZoom] = useState(1)
    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→  const [isDragging, setIsDragging] = useState(false)
    58→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    59→  const [imageLoaded, setImageLoaded] = useState(false)
    60→  const containerRef = useRef<HTMLDivElement>(null)
    61→  const lastTouchDist = useRef<number | null>(null)
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) return
    70→    const handleKeyDown = (e: KeyboardEvent) => {
    71→      if (e.key === 'Escape') onClose()
    72→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    73→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    74→      else if (e.key === '0') resetView()
    75→    }
    76→    window.addEventListener('keydown', handleKeyDown)
    77→    return () => window.removeEventListener('keydown', handleKeyDown)
    78→  }, [isOpen, onClose, resetView])
    79→
    80→  const handleWheel = useCallback((e: React.WheelEvent) => {
    81→    e.preventDefault()
    82→    setZoom(z => {
    83→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    84→      if (next <= 1) setPosition({ x: 0, y: 0 })
    85→      return next
    86→    })
    87→  }, [])
    88→
    89→  const handleMouseDown = (e: React.MouseEvent) => {
    90→    if (zoom <= 1) return
    91→    setIsDragging(true)
    92→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    93→  }
    94→
    95→  const handleMouseMove = (e: React.MouseEvent) => {
    96→    if (!isDragging) return
    97→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
    98→  }
    99→
   100→  const handleMouseUp = () => setIsDragging(false)
   101→
   102→  const handleTouchStart = (e: React.TouchEvent) => {
   103→    if (e.touches.length === 2) {
   104→      const dx = e.touches[0].clientX - e.touches[1].clientX
   105→      const dy = e.touches[0].clientY - e.touches[1].clientY
   106→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   107→    } else if (e.touches.length === 1 && zoom > 1) {
   108→      setIsDragging(true)
   109→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   110→    }
   111→  }
   112→
   113→  const handleTouchMove = (e: React.TouchEvent) => {
   114→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   115→      const dx = e.touches[0].clientX - e.touches[1].clientX
   116→      const dy = e.touches[0].clientY - e.touches[1].clientY
   117→      const dist = Math.sqrt(dx * dx + dy * dy)
   118→      const scale = dist / lastTouchDist.current
   119→      setZoom(z => {
   120→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   121→        if (next <= 1) setPosition({ x: 0, y: 0 })
   122→        return next
   123→      })
   124→      lastTouchDist.current = dist
   125→    } else if (e.touches.length === 1 && isDragging) {
   126→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   127→    }
   128→  }
   129→
   130→  const handleTouchEnd = () => {
   131→    lastTouchDist.current = null
   132→    setIsDragging(false)
   133→  }
   134→
   135→  if (!isOpen) return null
   136→
   137→  return createPortal(
   138→    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col items-center justify-center">
   139→      <div
   140→        ref={containerRef}
   141→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   142→        onWheel={handleWheel}
   143→        onMouseDown={handleMouseDown}
   144→        onMouseMove={handleMouseMove}
   145→        onMouseUp={handleMouseUp}
   146→        onMouseLeave={handleMouseUp}
   147→        onTouchStart={handleTouchStart}
   148→        onTouchMove={handleTouchMove}
   149→        onTouchEnd={handleTouchEnd}
   150→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   151→      >
   152→        {!imageLoaded && (
   153→          <Loader2 className="size-10 text-white animate-spin" />
   154→        )}
   155→        <img
   156→          src={imageUrl}
   157→          alt={label}
   158→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   159→          style={{
   160→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   161→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   162→            display: imageLoaded ? 'block' : 'none',
   163→          }}
   164→          onLoad={() => setImageLoaded(true)}
   165→          draggable={false}
   166→        />
   167→      </div>
   168→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   169→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   170→          <ZoomOut className="size-5" />
   171→        </Button>
   172→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   173→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   174→          <ZoomIn className="size-5" />
   175→        </Button>
   176→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   177→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   178→        </Button>
   179→        <span className="text-white/70 text-sm ml-2">{label}</span>
   180→      </div>
   181→    </div>,
   182→    document.body
   183→  )
   184→}

   186→// ===== Exercise Detail Dialog =====
   187→function ExerciseDetailDialog({
   188→  exercise,
   189→  chapter,
   190→  exercises,
   191→  onClose,
   192→}: {
   193→  exercise: ExerciseData
   194→  chapter: ChapterData
   195→  exercises: ExerciseData[]
   196→  onClose: () => void
   197→}) {
   198→  const { user, setSelectedExercise } = useAppStore()
   199→  const student = user as StudentUser
   200→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   201→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   202→  const [saving, setSaving] = useState(false)
   203→  const [comments, setComments] = useState<CommentData[]>([])
   204→  const [newComment, setNewComment] = useState('')
   205→  const [sendingComment, setSendingComment] = useState(false)
   206→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   207→
   208→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   209→  const totalExercises = exercises.length
   210→
   211→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   212→
   213→  useEffect(() => {
   214→    let cancelled = false
   215→    if (commentsLoaded) return
   216→    const load = async () => {
   217→      try {
   218→        const res = await fetch('/api/comments')
   219→        if (res.ok && !cancelled) {
   220→          const data = await res.json()
   221→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   222→        }
   223→      } catch { /* ignore */ }
   224→      if (!cancelled) setCommentsLoaded(true)
   225→    }
   226→    load()
   227→  }, [exercise.id, commentsLoaded])
   228→
   229→  const refetchComments = async () => {
   230→    try {
   231→      const res = await fetch('/api/comments')
   232→      if (res.ok) {
   233→        const data = await res.json()
   234→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   235→      }
   236→    } catch { /* ignore */ }
   237→  }
   238→
   239→  const handleSave = async () => {
   240→    setSaving(true)
   241→    try {
   242→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   243→        method: 'PUT',
   244→        headers: { 'Content-Type': 'application/json' },
   245→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   246→      })
   247→      toast.success('Progression sauvegardée !')
   248→    } catch {
   249→      toast.error('Erreur lors de la sauvegarde.')
   250→    }
   251→    setSaving(false)
   252→  }
   253→
   254→  const handleSendComment = async () => {
   255→    if (!newComment.trim()) return
   256→    setSendingComment(true)
   257→    try {
   258→      await fetch('/api/comments', {
   259→        method: 'POST',
   260→        headers: { 'Content-Type': 'application/json' },
   261→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   262→      })
   263→      setNewComment('')
   264→      refetchComments()
   265→    } catch {
   266→      toast.error("Erreur lors de l'envoi.")
   267→    }
   268→    setSendingComment(false)
   269→  }
   270→
   271→  const goToExercise = (direction: 'prev' | 'next') => {
   272→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   273→    if (newIndex < 0 || newIndex >= totalExercises) return
   274→    setSelectedExercise(exercises[newIndex])
   275→  }
   276→
   277→  const imageUrl = exercise.pageStart ? exerciseImageUrl(chapter.number, exercise.number) : ''
   278→
   279→  return (
   280→    <>
   281→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   282→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   283→          <DialogHeader>
   284→            <DialogTitle className="text-lg">
   285→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   286→            </DialogTitle>
   287→            <DialogDescription className="sr-only">
   288→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   289→            </DialogDescription>
   290→          </DialogHeader>
   291→
   292→          {imageUrl && (
   293→            <div className="space-y-2">
   294→              <div className="relative rounded-lg overflow-hidden border bg-gray-50 max-h-64">
   295→                <img src={imageUrl} alt={`Exercice ${exercise.number}`} className="w-full h-full object-contain max-h-64" />
   296→              </div>
   297→              <Button variant="outline" size="sm" onClick={() => setLightbox({ isOpen: true, imageUrl, label: `Ex.${exercise.number}` })}>
   298→                <Eye className="size-4 mr-1" /> Agrandir
   299→              </Button>
   300→            </div>
   301→          )}
   302→
   303→          <div className="rounded-lg border bg-gray-50 p-4">
   304→            <p className="text-sm whitespace-pre-wrap leading-relaxed">{exercise.content}</p>
   305→          </div>
   306→
   307→          <div className="space-y-2">
   308→            <Label className="text-sm font-medium">Statut</Label>
   309→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   310→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   311→                <RadioGroupItem value="not_started" />
   312→                <Circle className="size-4 text-gray-500" />
   313→                <span className="text-sm text-gray-500">Non commencé</span>
   314→              </label>
   315→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   316→                <RadioGroupItem value="in_progress" />
   317→                <Clock className="size-4 text-amber-600" />
   318→                <span className="text-sm text-amber-700">En cours</span>
   319→              </label>
   320→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   321→                <RadioGroupItem value="completed" />
   322→                <CheckCircle className="size-4 text-emerald-600" />
   323→                <span className="text-sm text-emerald-700">Terminé</span>
   324→              </label>
   325→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   326→                <RadioGroupItem value="difficulty" />
   327→                <AlertTriangle className="size-4 text-red-600" />
   328→                <span className="text-sm text-red-700">Difficulté</span>
   329→              </label>
   330→            </RadioGroup>
   331→          </div>
   332→
   333→          <div className="space-y-2">
   334→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   335→            <Textarea
   336→              id="notes"
   337→              value={note}
   338→              onChange={(e) => setNote(e.target.value)}
   339→              placeholder="Ajoutez vos notes ici..."
   340→              rows={3}
   341→            />
   342→          </div>
   343→
   344→          <div className="flex justify-end">
   345→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   346→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   347→              Sauvegarder
   348→            </Button>
   349→          </div>
   350→
   351→          <Separator />
   352→
   353→          <div className="space-y-3">
   354→            <h4 className="font-medium text-sm">Commentaires</h4>
   355→            {comments.map((comment) => (
   356→              <div key={comment.id} className="space-y-2">
   357→                <div className="flex items-start gap-2">
   358→                  <Avatar className="size-8">
   359→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   360→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   361→                    </AvatarFallback>
   362→                  </Avatar>
   363→                  <div className="flex-1">
   364→                    <div className="flex items-center gap-2">
   365→                      <span className="text-sm font-medium">
   366→                        {comment.student.firstName} {comment.student.lastName}
   367→                      </span>
   368→                      <span className="text-xs text-muted-foreground">
   369→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   370→                      </span>
   371→                    </div>
   372→                    <p className="text-sm mt-1">{comment.content}</p>
   373→                  </div>
   374→                </div>
   375→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   376→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   377→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   378→                    <span className="text-xs text-emerald-600 mt-1 block">
   379→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   380→                    </span>
   381→                  </div>
   382→                ))}
   383→              </div>
   384→            ))}
   385→            <div className="flex gap-2">
   386→              <Textarea
   387→                value={newComment}
   388→                onChange={(e) => setNewComment(e.target.value)}
   389→                placeholder="Ajouter un commentaire..."
   390→                rows={2}
   391→              />
   392→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   393→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   394→              </Button>
   395→            </div>
   396→          </div>
   397→
   398→          <DialogFooter className="flex-row justify-between">
   399→            <Button
   400→              variant="outline"
   401→              onClick={() => goToExercise('prev')}
   402→              disabled={currentIndex <= 0}
   403→            >
   404→              <ChevronLeft className="size-4 mr-1" /> Précédent
   405→            </Button>
   406→            <Button
   407→              variant="outline"
   408→              onClick={() => goToExercise('next')}
   409→              disabled={currentIndex >= totalExercises - 1}
   410→            >
   411→              Suivant <ChevronRight className="size-4 ml-1" />
   412→            </Button>
   413→          </DialogFooter>
   414→        </DialogContent>
   415→      </Dialog>
   416→
   417→      <ImageLightbox
   418→        key={lightbox.imageUrl}
   419→        isOpen={lightbox.isOpen}
   420→        imageUrl={lightbox.imageUrl}
   421→        label={lightbox.label}
   422→        onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   423→      />
   424→    </>
   425→  )
   426→}

   478→      const res = await fetch('/api/student/register', {
   479→        method: 'POST',
   480→        headers: { 'Content-Type': 'application/json' },
   481→        body: JSON.stringify({
   482→          firstName: regFirstName, lastName: regLastName, email: regEmail,
   483→          className: regClassName, schoolName: regSchoolName, phone: regPhone, password: regPassword,
   484→        }),

   488→        setMode('login')
   489→      } else {
   490→        const data = await res.json().catch(() => ({}))

   494→      toast.error("Erreur lors de l'inscription.")
   495→    }
   496→    setLoading(false)
   497→  }

   499→  const handleTeacherLogin = async () => {
   500→    setLoading(true)
   501→    try {
   502→      const res = await fetch('/api/teacher/login', {

   821→function ChapterRowInline({ chapter, isExpanded, isLoading, exercises, onToggle, onExerciseClick, onImageClick }: {
   822→  chapter: ChapterData
   823→  isExpanded: boolean
   824→  isLoading: boolean
   825→  exercises: ExerciseData[]

   829→}) {
   830→  const allDone = chapter.completedCount === chapter.exerciseCount
   831→  return (
   832→    <div className="rounded-lg border border-l-4 border-l-emerald-400 overflow-hidden">
   833→      <button
   834→        onClick={onToggle}
   835→        className="w-full flex items-center gap-3 p-3 hover:bg-gray-50 transition-colors text-left"
   836→      >

   839→        <span className="flex-1 text-sm truncate">{chapter.title}</span>
