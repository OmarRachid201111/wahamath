    45→// ===== Image Lightbox (Portal) =====
    46→function ImageLightbox({
    47→  isOpen, imageUrl, label, onClose
    48→}: {
    49→  isOpen: boolean
    50→  imageUrl: string
    51→  label: string
    52→  onClose: () => void
    53→}) {
    54→  const [zoom, setZoom] = useState(1)
    55→  const [position, setPosition] = useState({ x: 0, y: 0 })
    56→  const [isDragging, setIsDragging] = useState(false)
    57→  const [imageLoaded, setImageLoaded] = useState(false)
    58→  const containerRef = useRef<HTMLDivElement>(null)
    59→  const lastTouchDist = useRef<number | null>(null)
    60→  const dragStartRef = useRef({ x: 0, y: 0 })
    61→  const posStartRef = useRef({ x: 0, y: 0 })
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) return
    70→    const handleKeyDown = (e: KeyboardEvent) => {
    71→      if (e.key === 'Escape') {
    72→        e.stopPropagation()
    73→        e.preventDefault()
    74→        onClose()
    75→      }
    76→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    77→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    78→      else if (e.key === '0') resetView()
    79→    }
    80→    window.addEventListener('keydown', handleKeyDown, true)
    81→    return () => window.removeEventListener('keydown', handleKeyDown, true)
    82→  }, [isOpen, onClose, resetView])
    83→
    84→  // Global mousemove/mouseup for smooth drag even when cursor leaves container
    85→  useEffect(() => {
    86→    if (!isDragging) return
    87→    const handleGlobalMove = (e: MouseEvent) => {
    88→      setPosition({
    89→        x: posStartRef.current.x + (e.clientX - dragStartRef.current.x),
    90→        y: posStartRef.current.y + (e.clientY - dragStartRef.current.y),
    91→      })
    92→    }
    93→    const handleGlobalUp = () => {
    94→      setIsDragging(false)
    95→    }
    96→    window.addEventListener('mousemove', handleGlobalMove)
    97→    window.addEventListener('mouseup', handleGlobalUp)
    98→    return () => {
    99→      window.removeEventListener('mousemove', handleGlobalMove)
   100→      window.removeEventListener('mouseup', handleGlobalUp)
   101→    }
   102→  }, [isDragging])
   103→
   104→  const handleWheel = useCallback((e: React.WheelEvent) => {
   105→    e.preventDefault()
   106→    setZoom(z => {
   107→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
   108→      if (next <= 1) setPosition({ x: 0, y: 0 })
   109→      return next
   110→    })
   111→  }, [])
   112→
   113→  const handlePointerDown = (e: React.PointerEvent) => {
   114→    if (zoom <= 1) return
   115→    e.preventDefault()
   116→    setIsDragging(true)
   117→    dragStartRef.current = { x: e.clientX, y: e.clientY }
   118→    posStartRef.current = { x: position.x, y: position.y }
   119→  }
   120→
   121→  const handleTouchStart = (e: React.TouchEvent) => {
   122→    if (e.touches.length === 2) {
   123→      const dx = e.touches[0].clientX - e.touches[1].clientX
   124→      const dy = e.touches[0].clientY - e.touches[1].clientY
   125→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   126→    } else if (e.touches.length === 1 && zoom > 1) {
   127→      setIsDragging(true)
   128→      dragStartRef.current = { x: e.touches[0].clientX, y: e.touches[0].clientY }
   129→      posStartRef.current = { x: position.x, y: position.y }
   130→    }
   131→  }
   132→
   133→  const handleTouchMove = (e: React.TouchEvent) => {
   134→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   135→      const dx = e.touches[0].clientX - e.touches[1].clientX
   136→      const dy = e.touches[0].clientY - e.touches[1].clientY
   137→      const dist = Math.sqrt(dx * dx + dy * dy)
   138→      const scale = dist / lastTouchDist.current
   139→      setZoom(z => {
   140→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   141→        if (next <= 1) setPosition({ x: 0, y: 0 })
   142→        return next
   143→      })
   144→      lastTouchDist.current = dist
   145→    } else if (e.touches.length === 1 && isDragging) {
   146→      setPosition({
   147→        x: posStartRef.current.x + (e.touches[0].clientX - dragStartRef.current.x),
   148→        y: posStartRef.current.y + (e.touches[0].clientY - dragStartRef.current.y),
   149→      })
   150→    }
   151→  }
   152→
   153→  const handleTouchEnd = () => {
   154→    lastTouchDist.current = null
   155→    setIsDragging(false)
   156→  }
   157→
   158→
   159→  // Close lightbox when clicking the dark overlay area (not the image or controls)
   160→  const handleOverlayClick = useCallback((e: React.MouseEvent) => {
   161→    // Only close if clicking directly on the overlay, not on children
   162→    if (e.target === e.currentTarget) {
   163→      onClose()
   164→    }
   165→  }, [onClose])
   166→
   167→  if (!isOpen) return null
   168→
   169→  return (
   170→    <div
   171→      className="fixed inset-0 z-[9999] bg-black/90 flex flex-col items-center justify-center"
   172→      // Stop ALL event propagation so nothing reaches Radix Dialog handlers on document
   173→      onPointerDown={(e) => e.stopImmediatePropagation()}
   174→      onPointerUp={(e) => e.stopImmediatePropagation()}
   175→      onMouseDown={(e) => e.stopImmediatePropagation()}
   176→      onMouseUp={(e) => e.stopImmediatePropagation()}
   177→      onClick={handleOverlayClick}
   178→    >
   179→      {/* Close button */}
   180→      <button
   181→        onClick={(e) => { e.stopPropagation(); onClose() }}
   182→        className="absolute top-4 right-4 z-10 text-white/70 hover:text-white transition-colors p-2 rounded-full hover:bg-white/10"
   183→        aria-label="Fermer"
   184→      >
   185→        <X className="size-6" />
   186→      </button>
   187→      <div
   188→        ref={containerRef}
   189→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   190→        onWheel={handleWheel}
   191→        onPointerDown={handlePointerDown}
   192→        onTouchStart={handleTouchStart}
   193→        onTouchMove={handleTouchMove}
   194→        onTouchEnd={handleTouchEnd}
   195→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default', userSelect: 'none' }}
   196→      >
   197→        {!imageLoaded && (
   198→          <Loader2 className="size-10 text-white animate-spin" />
   199→        )}
   200→        <img
   201→          src={imageUrl}
   202→          alt={label}
   203→          className="max-w-[90vw] max-h-[80vh] object-contain select-none pointer-events-none"
   204→          style={{
   205→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   206→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   207→            display: imageLoaded ? 'block' : 'none',
   208→          }}
   209→          onLoad={() => setImageLoaded(true)}
   210→          draggable={false}
   211→        />
   212→      </div>
   213→      {/* Zoom control bar — stopPropagation on the bar itself */}
   214→      <div
   215→        className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl"
   216→        onClick={(e) => e.stopPropagation()}
   217→      >
   218→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.max(z - 0.25, 0.5)) }}>
   219→          <ZoomOut className="size-5" />
   220→        </Button>
   221→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   222→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.min(z + 0.25, 5)) }}>
   223→          <ZoomIn className="size-5" />
   224→        </Button>
   225→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); resetView() }}>
   226→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   227→        </Button>
   228→        <span className="text-white/70 text-sm ml-2">{label}</span>
   229→      </div>
   230→    </div>
   231→  )
   232→}

   234→// ===== Exercise Detail Dialog =====
   235→function ExerciseDetailDialog({
   236→  exercise,
   237→  chapter,
   238→  exercises,
   239→  onClose,
   240→}: {
   241→  exercise: ExerciseData
   242→  chapter: ChapterData
   243→  exercises: ExerciseData[]
   244→  onClose: () => void
   245→}) {
   246→  const { user, setSelectedExercise, openLightbox, lightbox } = useAppStore()
   247→  const student = user as StudentUser
   248→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   249→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   250→  const [saving, setSaving] = useState(false)
   251→  const [comments, setComments] = useState<CommentData[]>([])
   252→  const [newComment, setNewComment] = useState('')
   253→  const [sendingComment, setSendingComment] = useState(false)
   254→
   255→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   256→  const totalExercises = exercises.length
   257→
   258→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   259→
   260→  useEffect(() => {
   261→    let cancelled = false
   262→    if (commentsLoaded) return
   263→    const load = async () => {
   264→      try {
   265→        const res = await fetch('/api/comments')
   266→        if (res.ok && !cancelled) {
   267→          const data = await res.json()
   268→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   269→        }
   270→      } catch { /* ignore */ }
   271→      if (!cancelled) setCommentsLoaded(true)
   272→    }
   273→    load()
   274→  }, [exercise.id, commentsLoaded])
   275→
   276→  const refetchComments = async () => {
   277→    try {
   278→      const res = await fetch('/api/comments')
   279→      if (res.ok) {
   280→        const data = await res.json()
   281→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   282→      }
   283→    } catch { /* ignore */ }
   284→  }
   285→
   286→  const handleSave = async () => {
   287→    setSaving(true)
   288→    try {
   289→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   290→        method: 'PUT',
   291→        headers: { 'Content-Type': 'application/json' },
   292→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   293→      })
   294→      toast.success('Progression sauvegardée !')
   295→    } catch {
   296→      toast.error('Erreur lors de la sauvegarde.')
   297→    }
   298→    setSaving(false)
   299→  }
   300→
   301→  const handleSendComment = async () => {
   302→    if (!newComment.trim()) return
   303→    setSendingComment(true)
   304→    try {
   305→      await fetch('/api/comments', {
   306→        method: 'POST',
   307→        headers: { 'Content-Type': 'application/json' },
   308→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   309→      })
   310→      setNewComment('')
   311→      refetchComments()
   312→    } catch {
   313→      toast.error("Erreur lors de l'envoi.")
   314→    }
   315→    setSendingComment(false)
   316→  }
   317→
   318→  const goToExercise = (direction: 'prev' | 'next') => {
   319→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   320→    if (newIndex < 0 || newIndex >= totalExercises) return
   321→    setSelectedExercise(exercises[newIndex])
   322→  }
   323→
   324→  // Build list of page images for this exercise
   325→  const pageImages: { url: string; pageNum: number }[] = []
   326→  if (exercise.pageStart) {
   327→    const start = exercise.pageStart
   328→    const end = exercise.pageEnd || start
   329→    for (let p = start; p <= end; p++) {
   330→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   331→    }
   332→  }
   333→
   334→  return (
   335→      <Dialog open onOpenChange={(open) => { if (!open && !lightbox.isOpen) onClose() }}>
   336→        <DialogContent 
   337→          className="max-w-4xl max-h-[90vh] overflow-y-auto"
   338→        >
   339→          <DialogHeader>
   340→            <DialogTitle className="text-lg">
   341→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   342→            </DialogTitle>
   343→            <DialogDescription className="sr-only">
   344→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   345→            </DialogDescription>
   346→          </DialogHeader>
   347→
   348→          {pageImages.length > 0 && (
   349→            <div className="space-y-3">
   350→              {pageImages.map((img) => (
   351→                <div key={img.pageNum} className="space-y-1.5">
   352→                  <div
   353→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   354→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   355→                  >
   356→                    <img
   357→                      src={img.url}
   358→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   359→                      className="w-full h-auto object-contain"
   360→                      loading="lazy"
   361→                    />
   362→                  </div>
   363→                  <Button
   364→                    variant="outline"
   365→                    size="sm"
   366→                    className="w-full"
   367→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   368→                  >
   369→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   370→                  </Button>
   371→                </div>
   372→              ))}
   373→            </div>
   374→          )}
   375→
   376→          <div className="space-y-2">
   377→            <Label className="text-sm font-medium">Statut</Label>
   378→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   379→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   380→                <RadioGroupItem value="not_started" />
   381→                <Circle className="size-4 text-gray-500" />
   382→                <span className="text-sm text-gray-500">Non commencé</span>
   383→              </label>
   384→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   385→                <RadioGroupItem value="in_progress" />
   386→                <Clock className="size-4 text-amber-600" />
   387→                <span className="text-sm text-amber-700">En cours</span>
   388→              </label>
   389→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   390→                <RadioGroupItem value="completed" />
   391→                <CheckCircle className="size-4 text-emerald-600" />
   392→                <span className="text-sm text-emerald-700">Terminé</span>
   393→              </label>
   394→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   395→                <RadioGroupItem value="difficulty" />
   396→                <AlertTriangle className="size-4 text-red-600" />
   397→                <span className="text-sm text-red-700">Difficulté</span>
   398→              </label>
   399→            </RadioGroup>
   400→          </div>
   401→
   402→          <div className="space-y-2">
   403→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   404→            <Textarea
   405→              id="notes"
   406→              value={note}
   407→              onChange={(e) => setNote(e.target.value)}
   408→              placeholder="Ajoutez vos notes ici..."
   409→              rows={3}
   410→            />
   411→          </div>
   412→
   413→          <div className="flex justify-end">
   414→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   415→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   416→              Sauvegarder
   417→            </Button>
   418→          </div>
   419→
   420→          <Separator />
   421→
   422→          <div className="space-y-3">
   423→            <h4 className="font-medium text-sm">Commentaires</h4>
   424→            {comments.map((comment) => (
   425→              <div key={comment.id} className="space-y-2">
   426→                <div className="flex items-start gap-2">
   427→                  <Avatar className="size-8">
   428→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   429→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   430→                    </AvatarFallback>
   431→                  </Avatar>
   432→                  <div className="flex-1">
   433→                    <div className="flex items-center gap-2">
   434→                      <span className="text-sm font-medium">
   435→                        {comment.student.firstName} {comment.student.lastName}
   436→                      </span>
   437→                      <span className="text-xs text-muted-foreground">
   438→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   439→                      </span>
   440→                    </div>
   441→                    <p className="text-sm mt-1">{comment.content}</p>
   442→                  </div>
   443→                </div>
   444→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   445→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   446→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   447→                    <span className="text-xs text-emerald-600 mt-1 block">
   448→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   449→                    </span>
   450→                  </div>
   451→                ))}
   452→              </div>
   453→            ))}
   454→            <div className="flex gap-2">
   455→              <Textarea
   456→                value={newComment}
   457→                onChange={(e) => setNewComment(e.target.value)}
   458→                placeholder="Ajouter un commentaire..."
   459→                rows={2}
   460→              />
   461→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   462→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   463→              </Button>
   464→            </div>
   465→          </div>
   466→
   467→          <DialogFooter className="flex-row justify-between">
   468→            <Button
   469→              variant="outline"
   470→              onClick={() => goToExercise('prev')}
   471→              disabled={currentIndex <= 0}
   472→            >
   473→              <ChevronLeft className="size-4 mr-1" /> Précédent
   474→            </Button>
   475→            <Button
   476→              variant="outline"
   477→              onClick={() => goToExercise('next')}
   478→              disabled={currentIndex >= totalExercises - 1}
   479→            >
   480→              Suivant <ChevronRight className="size-4 ml-1" />
   481→            </Button>
   482→          </DialogFooter>
   483→
   484→        </DialogContent>
   485→      </Dialog>
   486→  )
   487→}

   770→      const res = await fetch(`/api/chapters/${chapter.id}/exercises?studentId=${student.id}`)
   771→      if (res.ok) {
   772→        const data = await res.json()
   773→        setChapterExercises(prev => ({ ...prev, [chapter.id]: data.exercises || [] }))

   808→                      isExpanded={expandedChapterId === ch.id}
   809→                      isLoading={loadingExercises === ch.id}
   810→                      exercises={chapterExercises[ch.id] || []}
   811→                      onToggle={() => toggleChapter(ch)}
   812→                      onExerciseClick={(ex) => setSelectedExercise(ex)}
   813→                      onImageClick={(ex) => {
   814→                        if (ex.pageStart) {
   815→                          openLightbox(pageImageUrl(ex.pageStart!), `Ex.${ex.number}`)
   816→                        }

   818→                    />
   819→                  ))}
   820→                </div>
   821→              </AccordionContent>
   822→            </AccordionItem>
   823→          )}
   824→
   825→          {algebre.length > 0 && (

   873→function ChapterRowInline({ chapter, isExpanded, isLoading, exercises, onToggle, onExerciseClick, onImageClick }: {

   876→  isLoading: boolean

   881→}) {

  1560→  useEffect(() => {

  1568→      } catch { /* ignore */ }

  1570→    fetchProfile()
  1571→  }, [])

  1573→  const displayTeacher = profile || teacher
  1574→

  1591→      const res = await fetch('/api/teacher/profile', {

  1594→        body: JSON.stringify({ currentPassword, newPassword }),

  1600→        setConfirmPassword('')
  1601→      } else {
  1602→        setError('Mot de passe actuel incorrect.')
  1603→      }
  1604→    } catch {
  1605→      setError('Erreur lors de la modification.')
  1606→    }
  1607→    setSaving(false)
