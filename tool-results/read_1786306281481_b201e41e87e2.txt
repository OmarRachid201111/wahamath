    45→// ===== Image Lightbox (Portal) =====
    46→function ImageLightbox({
    47→  isOpen, imageUrl, label, onClose
    48→}: {
    49→  isOpen: boolean
    50→  imageUrl: string
    51→  label: string
    52→  onClose: () => void
    53→}) {
    54→  const [zoom, setZoom] = useState(1)
    55→  const [position, setPosition] = useState({ x: 0, y: 0 })
    56→  const [isDragging, setIsDragging] = useState(false)
    57→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    58→  const [imageLoaded, setImageLoaded] = useState(false)
    59→  const containerRef = useRef<HTMLDivElement>(null)
    60→  const lastTouchDist = useRef<number | null>(null)
    61→
    62→  const resetView = useCallback(() => {
    63→    setZoom(1)
    64→    setPosition({ x: 0, y: 0 })
    65→  }, [])
    66→
    67→  useEffect(() => {
    68→    if (!isOpen) return
    69→    const handleKeyDown = (e: KeyboardEvent) => {
    70→      if (e.key === 'Escape') {
    71→        e.stopPropagation()
    72→        e.preventDefault()
    73→        onClose()
    74→      }
    75→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    76→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    77→      else if (e.key === '0') resetView()
    78→    }
    79→    window.addEventListener('keydown', handleKeyDown, true)
    80→    return () => window.removeEventListener('keydown', handleKeyDown, true)
    81→  }, [isOpen, onClose, resetView])
    82→
    83→  const handleWheel = useCallback((e: React.WheelEvent) => {
    84→    e.preventDefault()
    85→    setZoom(z => {
    86→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    87→      if (next <= 1) setPosition({ x: 0, y: 0 })
    88→      return next
    89→    })
    90→  }, [])
    91→
    92→  const handleMouseDown = (e: React.MouseEvent) => {
    93→    if (zoom <= 1) return
    94→    setIsDragging(true)
    95→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    96→  }
    97→
    98→  const handleMouseMove = (e: React.MouseEvent) => {
    99→    if (!isDragging) return
   100→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
   101→  }
   102→
   103→  const handleMouseUp = () => setIsDragging(false)
   104→
   105→  const handleTouchStart = (e: React.TouchEvent) => {
   106→    if (e.touches.length === 2) {
   107→      const dx = e.touches[0].clientX - e.touches[1].clientX
   108→      const dy = e.touches[0].clientY - e.touches[1].clientY
   109→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   110→    } else if (e.touches.length === 1 && zoom > 1) {
   111→      setIsDragging(true)
   112→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   113→    }
   114→  }
   115→
   116→  const handleTouchMove = (e: React.TouchEvent) => {
   117→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   118→      const dx = e.touches[0].clientX - e.touches[1].clientX
   119→      const dy = e.touches[0].clientY - e.touches[1].clientY
   120→      const dist = Math.sqrt(dx * dx + dy * dy)
   121→      const scale = dist / lastTouchDist.current
   122→      setZoom(z => {
   123→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   124→        if (next <= 1) setPosition({ x: 0, y: 0 })
   125→        return next
   126→      })
   127→      lastTouchDist.current = dist
   128→    } else if (e.touches.length === 1 && isDragging) {
   129→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   130→    }
   131→  }
   132→
   133→  const handleTouchEnd = () => {
   134→    lastTouchDist.current = null
   135→    setIsDragging(false)
   136→  }
   137→
   138→  if (!isOpen) return null
   139→
   140→  return (
   141→    <div
   142→      className="fixed inset-0 z-[9999] bg-black/90 flex flex-col items-center justify-center"
   143→      onClick={(e) => e.stopPropagation()}
   144→      onMouseDown={(e) => e.stopPropagation()}
   145→    >
   146→      <div
   147→        ref={containerRef}
   148→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   149→        onWheel={handleWheel}
   150→        onMouseDown={handleMouseDown}
   151→        onMouseMove={handleMouseMove}
   152→        onMouseUp={handleMouseUp}
   153→        onMouseLeave={handleMouseUp}
   154→        onTouchStart={handleTouchStart}
   155→        onTouchMove={handleTouchMove}
   156→        onTouchEnd={handleTouchEnd}
   157→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   158→      >
   159→        {!imageLoaded && (
   160→          <Loader2 className="size-10 text-white animate-spin" />
   161→        )}
   162→        <img
   163→          src={imageUrl}
   164→          alt={label}
   165→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   166→          style={{
   167→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   168→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   169→            display: imageLoaded ? 'block' : 'none',
   170→          }}
   171→          onLoad={() => setImageLoaded(true)}
   172→          draggable={false}
   173→        />
   174→      </div>
   175→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   176→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   177→          <ZoomOut className="size-5" />
   178→        </Button>
   179→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   180→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   181→          <ZoomIn className="size-5" />
   182→        </Button>
   183→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   184→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   185→        </Button>
   186→        <span className="text-white/70 text-sm ml-2">{label}</span>
   187→      </div>
   188→    </div>
   189→  )
   190→}

   192→// ===== Exercise Detail Dialog =====
   193→function ExerciseDetailDialog({
   194→  exercise,
   195→  chapter,
   196→  exercises,
   197→  onClose,
   198→}: {
   199→  exercise: ExerciseData
   200→  chapter: ChapterData
   201→  exercises: ExerciseData[]
   202→  onClose: () => void
   203→}) {
   204→  const { user, setSelectedExercise } = useAppStore()
   205→  const student = user as StudentUser
   206→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   207→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   208→  const [saving, setSaving] = useState(false)
   209→  const [comments, setComments] = useState<CommentData[]>([])
   210→  const [newComment, setNewComment] = useState('')
   211→  const [sendingComment, setSendingComment] = useState(false)
   212→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   213→
   214→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   215→  const totalExercises = exercises.length
   216→
   217→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   218→
   219→  useEffect(() => {
   220→    let cancelled = false
   221→    if (commentsLoaded) return
   222→    const load = async () => {
   223→      try {
   224→        const res = await fetch('/api/comments')
   225→        if (res.ok && !cancelled) {
   226→          const data = await res.json()
   227→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   228→        }
   229→      } catch { /* ignore */ }
   230→      if (!cancelled) setCommentsLoaded(true)
   231→    }
   232→    load()
   233→  }, [exercise.id, commentsLoaded])
   234→
   235→  const refetchComments = async () => {
   236→    try {
   237→      const res = await fetch('/api/comments')
   238→      if (res.ok) {
   239→        const data = await res.json()
   240→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   241→      }
   242→    } catch { /* ignore */ }
   243→  }
   244→
   245→  const handleSave = async () => {
   246→    setSaving(true)
   247→    try {
   248→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   249→        method: 'PUT',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   252→      })
   253→      toast.success('Progression sauvegardée !')
   254→    } catch {
   255→      toast.error('Erreur lors de la sauvegarde.')
   256→    }
   257→    setSaving(false)
   258→  }
   259→
   260→  const handleSendComment = async () => {
   261→    if (!newComment.trim()) return
   262→    setSendingComment(true)
   263→    try {
   264→      await fetch('/api/comments', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   268→      })
   269→      setNewComment('')
   270→      refetchComments()
   271→    } catch {
   272→      toast.error("Erreur lors de l'envoi.")
   273→    }
   274→    setSendingComment(false)
   275→  }
   276→
   277→  const goToExercise = (direction: 'prev' | 'next') => {
   278→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   279→    if (newIndex < 0 || newIndex >= totalExercises) return
   280→    setSelectedExercise(exercises[newIndex])
   281→  }
   282→
   283→  // Build list of page images for this exercise
   284→  const pageImages: { url: string; pageNum: number }[] = []
   285→  if (exercise.pageStart) {
   286→    const start = exercise.pageStart
   287→    const end = exercise.pageEnd || start
   288→    for (let p = start; p <= end; p++) {
   289→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   290→    }
   291→  }
   292→
   293→  return (
   294→    <>
   295→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   296→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   297→          <DialogHeader>
   298→            <DialogTitle className="text-lg">
   299→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   300→            </DialogTitle>
   301→            <DialogDescription className="sr-only">
   302→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   303→            </DialogDescription>
   304→          </DialogHeader>
   305→
   306→          {pageImages.length > 0 && (
   307→            <div className="space-y-3">
   308→              {pageImages.map((img) => (
   309→                <div key={img.pageNum} className="space-y-1.5">
   310→                  <div
   311→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   312→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   313→                  >
   314→                    <img
   315→                      src={img.url}
   316→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   317→                      className="w-full h-auto object-contain"
   318→                      loading="lazy"
   319→                    />
   320→                  </div>
   321→                  <Button
   322→                    variant="outline"
   323→                    size="sm"
   324→                    className="w-full"
   325→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   326→                  >
   327→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   328→                  </Button>
   329→                </div>
   330→              ))}
   331→            </div>
   332→          )}
   333→
   334→          <div className="space-y-2">
   335→            <Label className="text-sm font-medium">Statut</Label>
   336→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   337→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   338→                <RadioGroupItem value="not_started" />
   339→                <Circle className="size-4 text-gray-500" />
   340→                <span className="text-sm text-gray-500">Non commencé</span>
   341→              </label>
   342→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   343→                <RadioGroupItem value="in_progress" />
   344→                <Clock className="size-4 text-amber-600" />
   345→                <span className="text-sm text-amber-700">En cours</span>
   346→              </label>
   347→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   348→                <RadioGroupItem value="completed" />
   349→                <CheckCircle className="size-4 text-emerald-600" />
   350→                <span className="text-sm text-emerald-700">Terminé</span>
   351→              </label>
   352→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   353→                <RadioGroupItem value="difficulty" />
   354→                <AlertTriangle className="size-4 text-red-600" />
   355→                <span className="text-sm text-red-700">Difficulté</span>
   356→              </label>
   357→            </RadioGroup>
   358→          </div>
   359→
   360→          <div className="space-y-2">
   361→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   362→            <Textarea
   363→              id="notes"
   364→              value={note}
   365→              onChange={(e) => setNote(e.target.value)}
   366→              placeholder="Ajoutez vos notes ici..."
   367→              rows={3}
   368→            />
   369→          </div>
   370→
   371→          <div className="flex justify-end">
   372→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   373→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   374→              Sauvegarder
   375→            </Button>
   376→          </div>
   377→
   378→          <Separator />
   379→
   380→          <div className="space-y-3">
   381→            <h4 className="font-medium text-sm">Commentaires</h4>
   382→            {comments.map((comment) => (
   383→              <div key={comment.id} className="space-y-2">
   384→                <div className="flex items-start gap-2">
   385→                  <Avatar className="size-8">
   386→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   387→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   388→                    </AvatarFallback>
   389→                  </Avatar>
   390→                  <div className="flex-1">
   391→                    <div className="flex items-center gap-2">
   392→                      <span className="text-sm font-medium">
   393→                        {comment.student.firstName} {comment.student.lastName}
   394→                      </span>
   395→                      <span className="text-xs text-muted-foreground">
   396→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   397→                      </span>
   398→                    </div>
   399→                    <p className="text-sm mt-1">{comment.content}</p>
   400→                  </div>
   401→                </div>
   402→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   403→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   404→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   405→                    <span className="text-xs text-emerald-600 mt-1 block">
   406→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   407→                    </span>
   408→                  </div>
   409→                ))}
   410→              </div>
   411→            ))}
   412→            <div className="flex gap-2">
   413→              <Textarea
   414→                value={newComment}
   415→                onChange={(e) => setNewComment(e.target.value)}
   416→                placeholder="Ajouter un commentaire..."
   417→                rows={2}
   418→              />
   419→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   420→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   421→              </Button>
   422→            </div>
   423→          </div>
   424→
   425→          <DialogFooter className="flex-row justify-between">
   426→            <Button
   427→              variant="outline"
   428→              onClick={() => goToExercise('prev')}
   429→              disabled={currentIndex <= 0}
   430→            >
   431→              <ChevronLeft className="size-4 mr-1" /> Précédent
   432→            </Button>
   433→            <Button
   434→              variant="outline"
   435→              onClick={() => goToExercise('next')}
   436→              disabled={currentIndex >= totalExercises - 1}
   437→            >
   438→              Suivant <ChevronRight className="size-4 ml-1" />
   439→            </Button>
   440→          </DialogFooter>
   441→
   442→        </DialogContent>
   443→      </Dialog>
   444→
   445→      <ImageLightbox
   446→        key={lightbox.imageUrl}
   447→        isOpen={lightbox.isOpen}
   448→        imageUrl={lightbox.imageUrl}
   449→        label={lightbox.label}
   450→        onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   451→      />
   452→    </>
   453→  )
   454→}

   467→  // Register form

   469→  const [regLastName, setRegLastName] = useState('')
   470→  const [regEmail, setRegEmail] = useState('')
   471→  const [regClassName, setRegClassName] = useState('')

   494→        const data = await res.json().catch(() => ({}))

   497→    } catch {
   498→      toast.error('Erreur de connexion.')
   499→    }
   500→    setLoading(false)
   501→  }
   502→
   503→  const handleRegister = async () => {
   504→    setLoading(true)
   505→    try {
   506→      const res = await fetch('/api/student/register', {
   507→        method: 'POST',
   508→        headers: { 'Content-Type': 'application/json' },

   849→function ChapterRowInline({ chapter, isExpanded, isLoading, exercises, onToggle, onExerciseClick, onImageClick }: {
   850→  chapter: ChapterData
   851→  isExpanded: boolean
   852→  isLoading: boolean
   853→  exercises: ExerciseData[]

   855→  onExerciseClick: (ex: ExerciseData) => void
   856→  onImageClick: (ex: ExerciseData) => void
   857→}) {
   858→  const allDone = chapter.completedCount === chapter.exerciseCount
   859→  return (
   860→    <div className="rounded-lg border border-l-4 border-l-emerald-400 overflow-hidden">
   861→      <button
   862→        onClick={onToggle}
   863→        className="w-full flex items-center gap-3 p-3 hover:bg-gray-50 transition-colors text-left"
