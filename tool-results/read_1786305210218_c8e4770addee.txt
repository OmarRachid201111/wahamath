    46→// ===== Image Lightbox (Portal) =====
    47→function ImageLightbox({
    48→  isOpen, imageUrl, label, onClose
    49→}: {
    50→  isOpen: boolean
    51→  imageUrl: string
    52→  label: string
    53→  onClose: () => void
    54→}) {
    55→  const [zoom, setZoom] = useState(1)
    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→  const [isDragging, setIsDragging] = useState(false)
    58→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 })
    59→  const [imageLoaded, setImageLoaded] = useState(false)
    60→  const containerRef = useRef<HTMLDivElement>(null)
    61→  const lastTouchDist = useRef<number | null>(null)
    62→
    63→  const resetView = useCallback(() => {
    64→    setZoom(1)
    65→    setPosition({ x: 0, y: 0 })
    66→  }, [])
    67→
    68→  useEffect(() => {
    69→    if (!isOpen) return
    70→    const handleKeyDown = (e: KeyboardEvent) => {
    71→      if (e.key === 'Escape') onClose()
    72→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    73→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    74→      else if (e.key === '0') resetView()
    75→    }
    76→    window.addEventListener('keydown', handleKeyDown)
    77→    return () => window.removeEventListener('keydown', handleKeyDown)
    78→  }, [isOpen, onClose, resetView])
    79→
    80→  const handleWheel = useCallback((e: React.WheelEvent) => {
    81→    e.preventDefault()
    82→    setZoom(z => {
    83→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
    84→      if (next <= 1) setPosition({ x: 0, y: 0 })
    85→      return next
    86→    })
    87→  }, [])
    88→
    89→  const handleMouseDown = (e: React.MouseEvent) => {
    90→    if (zoom <= 1) return
    91→    setIsDragging(true)
    92→    setDragStart({ x: e.clientX - position.x, y: e.clientY - position.y })
    93→  }
    94→
    95→  const handleMouseMove = (e: React.MouseEvent) => {
    96→    if (!isDragging) return
    97→    setPosition({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y })
    98→  }
    99→
   100→  const handleMouseUp = () => setIsDragging(false)
   101→
   102→  const handleTouchStart = (e: React.TouchEvent) => {
   103→    if (e.touches.length === 2) {
   104→      const dx = e.touches[0].clientX - e.touches[1].clientX
   105→      const dy = e.touches[0].clientY - e.touches[1].clientY
   106→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   107→    } else if (e.touches.length === 1 && zoom > 1) {
   108→      setIsDragging(true)
   109→      setDragStart({ x: e.touches[0].clientX - position.x, y: e.touches[0].clientY - position.y })
   110→    }
   111→  }
   112→
   113→  const handleTouchMove = (e: React.TouchEvent) => {
   114→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   115→      const dx = e.touches[0].clientX - e.touches[1].clientX
   116→      const dy = e.touches[0].clientY - e.touches[1].clientY
   117→      const dist = Math.sqrt(dx * dx + dy * dy)
   118→      const scale = dist / lastTouchDist.current
   119→      setZoom(z => {
   120→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   121→        if (next <= 1) setPosition({ x: 0, y: 0 })
   122→        return next
   123→      })
   124→      lastTouchDist.current = dist
   125→    } else if (e.touches.length === 1 && isDragging) {
   126→      setPosition({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y })
   127→    }
   128→  }
   129→
   130→  const handleTouchEnd = () => {
   131→    lastTouchDist.current = null
   132→    setIsDragging(false)
   133→  }
   134→
   135→  if (!isOpen) return null
   136→
   137→  return createPortal(
   138→    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col items-center justify-center">
   139→      <div
   140→        ref={containerRef}
   141→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   142→        onWheel={handleWheel}
   143→        onMouseDown={handleMouseDown}
   144→        onMouseMove={handleMouseMove}
   145→        onMouseUp={handleMouseUp}
   146→        onMouseLeave={handleMouseUp}
   147→        onTouchStart={handleTouchStart}
   148→        onTouchMove={handleTouchMove}
   149→        onTouchEnd={handleTouchEnd}
   150→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default' }}
   151→      >
   152→        {!imageLoaded && (
   153→          <Loader2 className="size-10 text-white animate-spin" />
   154→        )}
   155→        <img
   156→          src={imageUrl}
   157→          alt={label}
   158→          className="max-w-[90vw] max-h-[80vh] object-contain select-none"
   159→          style={{
   160→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   161→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   162→            display: imageLoaded ? 'block' : 'none',
   163→          }}
   164→          onLoad={() => setImageLoaded(true)}
   165→          draggable={false}
   166→        />
   167→      </div>
   168→      <div className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl">
   169→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.max(z - 0.25, 0.5))}>
   170→          <ZoomOut className="size-5" />
   171→        </Button>
   172→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   173→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={() => setZoom(z => Math.min(z + 0.25, 5))}>
   174→          <ZoomIn className="size-5" />
   175→        </Button>
   176→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={resetView}>
   177→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   178→        </Button>
   179→        <span className="text-white/70 text-sm ml-2">{label}</span>
   180→      </div>
   181→    </div>,
   182→    document.body
   183→  )
   184→}

   186→// ===== Exercise Detail Dialog =====
   187→function ExerciseDetailDialog({
   188→  exercise,
   189→  chapter,
   190→  exercises,
   191→  onClose,
   192→}: {
   193→  exercise: ExerciseData
   194→  chapter: ChapterData
   195→  exercises: ExerciseData[]
   196→  onClose: () => void
   197→}) {
   198→  const { user, setSelectedExercise } = useAppStore()
   199→  const student = user as StudentUser
   200→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   201→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   202→  const [saving, setSaving] = useState(false)
   203→  const [comments, setComments] = useState<CommentData[]>([])
   204→  const [newComment, setNewComment] = useState('')
   205→  const [sendingComment, setSendingComment] = useState(false)
   206→  const [lightbox, setLightbox] = useState({ isOpen: false, imageUrl: '', label: '' })
   207→
   208→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   209→  const totalExercises = exercises.length
   210→
   211→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   212→
   213→  useEffect(() => {
   214→    let cancelled = false
   215→    if (commentsLoaded) return
   216→    const load = async () => {
   217→      try {
   218→        const res = await fetch('/api/comments')
   219→        if (res.ok && !cancelled) {
   220→          const data = await res.json()
   221→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   222→        }
   223→      } catch { /* ignore */ }
   224→      if (!cancelled) setCommentsLoaded(true)
   225→    }
   226→    load()
   227→  }, [exercise.id, commentsLoaded])
   228→
   229→  const refetchComments = async () => {
   230→    try {
   231→      const res = await fetch('/api/comments')
   232→      if (res.ok) {
   233→        const data = await res.json()
   234→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   235→      }
   236→    } catch { /* ignore */ }
   237→  }
   238→
   239→  const handleSave = async () => {
   240→    setSaving(true)
   241→    try {
   242→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   243→        method: 'PUT',
   244→        headers: { 'Content-Type': 'application/json' },
   245→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   246→      })
   247→      toast.success('Progression sauvegardée !')
   248→    } catch {
   249→      toast.error('Erreur lors de la sauvegarde.')
   250→    }
   251→    setSaving(false)
   252→  }
   253→
   254→  const handleSendComment = async () => {
   255→    if (!newComment.trim()) return
   256→    setSendingComment(true)
   257→    try {
   258→      await fetch('/api/comments', {
   259→        method: 'POST',
   260→        headers: { 'Content-Type': 'application/json' },
   261→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   262→      })
   263→      setNewComment('')
   264→      refetchComments()
   265→    } catch {
   266→      toast.error("Erreur lors de l'envoi.")
   267→    }
   268→    setSendingComment(false)
   269→  }
   270→
   271→  const goToExercise = (direction: 'prev' | 'next') => {
   272→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   273→    if (newIndex < 0 || newIndex >= totalExercises) return
   274→    setSelectedExercise(exercises[newIndex])
   275→  }
   276→
   277→  // Build list of page images for this exercise
   278→  const pageImages: { url: string; pageNum: number }[] = []
   279→  if (exercise.pageStart) {
   280→    const start = exercise.pageStart
   281→    const end = exercise.pageEnd || start
   282→    for (let p = start; p <= end; p++) {
   283→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   284→    }
   285→  }
   286→
   287→  return (
   288→    <>
   289→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   290→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   291→          <DialogHeader>
   292→            <DialogTitle className="text-lg">
   293→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   294→            </DialogTitle>
   295→            <DialogDescription className="sr-only">
   296→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   297→            </DialogDescription>
   298→          </DialogHeader>
   299→
   300→          {pageImages.length > 0 && (
   301→            <div className="space-y-3">
   302→              {pageImages.map((img) => (
   303→                <div key={img.pageNum} className="space-y-1.5">
   304→                  <div
   305→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   306→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   307→                  >
   308→                    <img
   309→                      src={img.url}
   310→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   311→                      className="w-full h-auto object-contain"
   312→                      loading="lazy"
   313→                    />
   314→                  </div>
   315→                  <Button
   316→                    variant="outline"
   317→                    size="sm"
   318→                    className="w-full"
   319→                    onClick={() => setLightbox({ isOpen: true, imageUrl: img.url, label: `Page ${img.pageNum}` })}
   320→                  >
   321→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   322→                  </Button>
   323→                </div>
   324→              ))}
   325→            </div>
   326→          )}
   327→
   328→          <div className="space-y-2">
   329→            <Label className="text-sm font-medium">Statut</Label>
   330→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   331→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   332→                <RadioGroupItem value="not_started" />
   333→                <Circle className="size-4 text-gray-500" />
   334→                <span className="text-sm text-gray-500">Non commencé</span>
   335→              </label>
   336→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   337→                <RadioGroupItem value="in_progress" />
   338→                <Clock className="size-4 text-amber-600" />
   339→                <span className="text-sm text-amber-700">En cours</span>
   340→              </label>
   341→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   342→                <RadioGroupItem value="completed" />
   343→                <CheckCircle className="size-4 text-emerald-600" />
   344→                <span className="text-sm text-emerald-700">Terminé</span>
   345→              </label>
   346→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   347→                <RadioGroupItem value="difficulty" />
   348→                <AlertTriangle className="size-4 text-red-600" />
   349→                <span className="text-sm text-red-700">Difficulté</span>
   350→              </label>
   351→            </RadioGroup>
   352→          </div>
   353→
   354→          <div className="space-y-2">
   355→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   356→            <Textarea
   357→              id="notes"
   358→              value={note}
   359→              onChange={(e) => setNote(e.target.value)}
   360→              placeholder="Ajoutez vos notes ici..."
   361→              rows={3}
   362→            />
   363→          </div>
   364→
   365→          <div className="flex justify-end">
   366→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   367→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   368→              Sauvegarder
   369→            </Button>
   370→          </div>
   371→
   372→          <Separator />
   373→
   374→          <div className="space-y-3">
   375→            <h4 className="font-medium text-sm">Commentaires</h4>
   376→            {comments.map((comment) => (
   377→              <div key={comment.id} className="space-y-2">
   378→                <div className="flex items-start gap-2">
   379→                  <Avatar className="size-8">
   380→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs">
   381→                      {comment.student.firstName[0]}{comment.student.lastName[0]}
   382→                    </AvatarFallback>
   383→                  </Avatar>
   384→                  <div className="flex-1">
   385→                    <div className="flex items-center gap-2">
   386→                      <span className="text-sm font-medium">
   387→                        {comment.student.firstName} {comment.student.lastName}
   388→                      </span>
   389→                      <span className="text-xs text-muted-foreground">
   390→                        {new Date(comment.createdAt).toLocaleDateString('fr-FR')}
   391→                      </span>
   392→                    </div>
   393→                    <p className="text-sm mt-1">{comment.content}</p>
   394→                  </div>
   395→                </div>
   396→                {comment.remarks.length > 0 && comment.remarks.map((remark) => (
   397→                  <div key={remark.id} className="ml-10 bg-emerald-50 rounded-lg p-3 border border-emerald-100">
   398→                    <p className="text-sm text-emerald-800">{remark.content}</p>
   399→                    <span className="text-xs text-emerald-600 mt-1 block">
   400→                      {new Date(remark.createdAt).toLocaleDateString('fr-FR')}
   401→                    </span>
   402→                  </div>
   403→                ))}
   404→              </div>
   405→            ))}
   406→            <div className="flex gap-2">
   407→              <Textarea
   408→                value={newComment}
   409→                onChange={(e) => setNewComment(e.target.value)}
   410→                placeholder="Ajouter un commentaire..."
   411→                rows={2}
   412→              />
   413→              <Button onClick={handleSendComment} disabled={sendingComment || !newComment.trim()} className="bg-emerald-600 hover:bg-emerald-700 self-end">
   414→                {sendingComment ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   415→              </Button>
   416→            </div>
   417→          </div>
   418→
   419→          <DialogFooter className="flex-row justify-between">
   420→            <Button
   421→              variant="outline"
   422→              onClick={() => goToExercise('prev')}
   423→              disabled={currentIndex <= 0}
   424→            >
   425→              <ChevronLeft className="size-4 mr-1" /> Précédent
   426→            </Button>
   427→            <Button
   428→              variant="outline"
   429→              onClick={() => goToExercise('next')}
   430→              disabled={currentIndex >= totalExercises - 1}
   431→            >
   432→              Suivant <ChevronRight className="size-4 ml-1" />
   433→            </Button>
   434→          </DialogFooter>
   435→        </DialogContent>
   436→      </Dialog>
   437→
   438→      <ImageLightbox
   439→        key={lightbox.imageUrl}
   440→        isOpen={lightbox.isOpen}
   441→        imageUrl={lightbox.imageUrl}
   442→        label={lightbox.label}
   443→        onClose={() => setLightbox({ isOpen: false, imageUrl: '', label: '' })}
   444→      />
   445→    </>
   446→  )
   447→}

   450→function AuthView() {
   451→  const { login } = useAppStore()
   452→  const [mode, setMode] = useState<'login' | 'register'>('login')
   453→  const [showTeacherLogin, setShowTeacherLogin] = useState(false)

   456→  // Login form
   457→  const [loginEmail, setLoginEmail] = useState('')
   458→  const [loginPassword, setLoginPassword] = useState('')
   459→
   460→  // Register form
   461→  const [regFirstName, setRegFirstName] = useState('')
   462→  const [regLastName, setRegLastName] = useState('')
   463→  const [regEmail, setRegEmail] = useState('')
   464→  const [regClassName, setRegClassName] = useState('')
   465→  const [regSchoolName, setRegSchoolName] = useState('')
   466→  const [regPhone, setRegPhone] = useState('')
   467→  const [regPassword, setRegPassword] = useState('')
   468→
   469→  // Teacher login
   470→  const [teacherPassword, setTeacherPassword] = useState('')
   471→
   472→  // Form states are initialized to defaults to prevent autofill
   473→
   474→  const handleLogin = async () => {
   475→    setLoading(true)
   476→    try {
   477→      const res = await fetch('/api/student/login', {
   478→        method: 'POST',
   479→        headers: { 'Content-Type': 'application/json' },
   480→        body: JSON.stringify({ email: loginEmail, password: loginPassword }),
   481→      })
   482→      if (res.ok) {
   483→        const student = await res.json()
   484→        toast.success(`Bienvenue ${student.firstName} !`)
   485→        login(student, 'student')
   486→      } else {
   487→        const data = await res.json().catch(() => ({}))
   488→        toast.error(data.error || 'Identifiants incorrects.')
   489→      }
   490→    } catch {
   491→      toast.error('Erreur de connexion.')
   492→    }
   493→    setLoading(false)
   494→  }
   495→
   496→  const handleRegister = async () => {
   497→    setLoading(true)
   498→    try {
   499→      const res = await fetch('/api/student/register', {
   500→        method: 'POST',
   501→        headers: { 'Content-Type': 'application/json' },
   502→        body: JSON.stringify({
   503→          firstName: regFirstName, lastName: regLastName, email: regEmail,
   504→          className: regClassName, schoolName: regSchoolName, phone: regPhone, password: regPassword,
   505→        }),

   774→                      onImageClick={(ex) => {
   775→                        if (ex.pageStart) {
   776→                          setLightbox({ isOpen: true, imageUrl: pageImageUrl(ex.pageStart!), label: `Ex.${ex.number}` })
   777→                        }
   778→                      }}

   780→                  ))}

   782→              </AccordionContent>

   842→function ChapterRowInline({ chapter, isExpanded, isLoading, exercises, onToggle, onExerciseClick, onImageClick }: {
   843→  chapter: ChapterData
   844→  isExpanded: boolean
   845→  isLoading: boolean
   846→  exercises: ExerciseData[]

   848→  onExerciseClick: (ex: ExerciseData) => void
   849→  onImageClick: (ex: ExerciseData) => void
   850→}) {
   851→  const allDone = chapter.completedCount === chapter.exerciseCount
   852→  return (
   853→    <div className="rounded-lg border border-l-4 border-l-emerald-400 overflow-hidden">
   854→      <button
   855→        onClick={onToggle}
   856→        className="w-full flex items-center gap-3 p-3 hover:bg-gray-50 transition-colors text-left"
   857→      >
   858→        <ChevronRight className={`size-4 text-muted-foreground transition-transform duration-200 ${isExpanded ? 'rotate-90' : ''}`} />
   859→        <span className="font-medium text-sm">Ch.{chapter.number}:</span>

  1531→function TeacherProfileView() {

  1533→  const teacher = user as TeacherUser
  1534→  const [profile, setProfile] = useState<TeacherUser | null>(null)

  1539→  const [error, setError] = useState('')
  1540→
  1541→  useEffect(() => {
  1542→    const fetchProfile = async () => {
  1543→      try {
  1544→        const res = await fetch('/api/teacher/profile')
  1545→        if (res.ok) {
  1546→          const data = await res.json()

  1549→      } catch { /* ignore */ }

  1551→    fetchProfile()
  1552→  }, [])
  1553→
  1554→  const displayTeacher = profile || teacher
  1555→
  1556→  const handleChangePassword = async () => {

  1559→      setError('Tous les champs sont requis.')

  1567→      setError('Les mots de passe ne correspondent pas.')
  1568→      return
  1569→    }
  1570→    setSaving(true)
  1571→    try {
  1572→      const res = await fetch('/api/teacher/profile', {

  1579→        setCurrentPassword('')
  1580→        setNewPassword('')
  1581→        setConfirmPassword('')
  1582→      } else {
  1583→        setError('Mot de passe actuel incorrect.')
  1584→      }
  1585→    } catch {
  1586→      setError('Erreur lors de la modification.')
  1587→    }
  1588→    setSaving(false)
  1589→  }

  1591→  return (
