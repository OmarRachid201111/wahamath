    46→    46→// ===== Image Lightbox (Portal) =====
    47→    47→function ImageLightbox({
    48→    48→  isOpen, imageUrl, label, onClose
    49→    49→}: {
    50→    50→  isOpen: boolean
    51→    51→  imageUrl: string
    52→    52→  label: string
    53→    53→  onClose: () => void
    54→    54→}) {
    55→    55→  const [zoom, setZoom] = useState(1)
    56→    56→  const [position, setPosition] = useState({ x: 0, y: 0 })
    57→    57→  const [isDragging, setIsDragging] = useState(false)
    58→    58→  const [imageLoaded, setImageLoaded] = useState(false)
    59→    59→  const containerRef = useRef<HTMLDivElement>(null)
    60→    60→  const lastTouchDist = useRef<number | null>(null)
    61→    61→  const dragStartRef = useRef({ x: 0, y: 0 })
    62→    62→  const posStartRef = useRef({ x: 0, y: 0 })
    63→    63→
    64→    64→  const resetView = useCallback(() => {
    65→    65→    setZoom(1)
    66→    66→    setPosition({ x: 0, y: 0 })
    67→    67→  }, [])
    68→    68→
    69→    69→  useEffect(() => {
    70→    70→    if (!isOpen) return
    71→    71→    const handleKeyDown = (e: KeyboardEvent) => {
    72→    72→      if (e.key === 'Escape') {
    73→    73→        e.stopPropagation()
    74→    74→        e.preventDefault()
    75→    75→        onClose()
    76→    76→      }
    77→    77→      else if (e.key === '+' || e.key === '=') setZoom(z => Math.min(z + 0.25, 5))
    78→    78→      else if (e.key === '-') setZoom(z => Math.max(z - 0.25, 0.5))
    79→    79→      else if (e.key === '0') resetView()
    80→    80→    }
    81→    81→    window.addEventListener('keydown', handleKeyDown, true)
    82→    82→    return () => window.removeEventListener('keydown', handleKeyDown, true)
    83→    83→  }, [isOpen, onClose, resetView])
    84→    84→
    85→    85→  // Global mousemove/mouseup for smooth drag even when cursor leaves container
    86→    86→  useEffect(() => {
    87→    87→    if (!isDragging) return
    88→    88→    const handleGlobalMove = (e: MouseEvent) => {
    89→    89→      setPosition({
    90→    90→        x: posStartRef.current.x + (e.clientX - dragStartRef.current.x),
    91→    91→        y: posStartRef.current.y + (e.clientY - dragStartRef.current.y),
    92→    92→      })
    93→    93→    }
    94→    94→    const handleGlobalUp = () => {
    95→    95→      setIsDragging(false)
    96→    96→    }
    97→    97→    window.addEventListener('mousemove', handleGlobalMove)
    98→    98→    window.addEventListener('mouseup', handleGlobalUp)
    99→    99→    return () => {
   100→   100→      window.removeEventListener('mousemove', handleGlobalMove)
   101→   101→      window.removeEventListener('mouseup', handleGlobalUp)
   102→   102→    }
   103→   103→  }, [isDragging])
   104→   104→
   105→   105→  const handleWheel = useCallback((e: React.WheelEvent) => {
   106→   106→    e.preventDefault()
   107→   107→    setZoom(z => {
   108→   108→      const next = e.deltaY < 0 ? Math.min(z + 0.15, 5) : Math.max(z - 0.15, 0.5)
   109→   109→      if (next <= 1) setPosition({ x: 0, y: 0 })
   110→   110→      return next
   111→   111→    })
   112→   112→  }, [])
   113→   113→
   114→   114→  const handlePointerDown = (e: React.PointerEvent) => {
   115→   115→    if (zoom <= 1) return
   116→   116→    e.preventDefault()
   117→   117→    setIsDragging(true)
   118→   118→    dragStartRef.current = { x: e.clientX, y: e.clientY }
   119→   119→    posStartRef.current = { x: position.x, y: position.y }
   120→   120→  }
   121→   121→
   122→   122→  const handleTouchStart = (e: React.TouchEvent) => {
   123→   123→    if (e.touches.length === 2) {
   124→   124→      const dx = e.touches[0].clientX - e.touches[1].clientX
   125→   125→      const dy = e.touches[0].clientY - e.touches[1].clientY
   126→   126→      lastTouchDist.current = Math.sqrt(dx * dx + dy * dy)
   127→   127→    } else if (e.touches.length === 1 && zoom > 1) {
   128→   128→      setIsDragging(true)
   129→   129→      dragStartRef.current = { x: e.touches[0].clientX, y: e.touches[0].clientY }
   130→   130→      posStartRef.current = { x: position.x, y: position.y }
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  const handleTouchMove = (e: React.TouchEvent) => {
   135→   135→    if (e.touches.length === 2 && lastTouchDist.current !== null) {
   136→   136→      const dx = e.touches[0].clientX - e.touches[1].clientX
   137→   137→      const dy = e.touches[0].clientY - e.touches[1].clientY
   138→   138→      const dist = Math.sqrt(dx * dx + dy * dy)
   139→   139→      const scale = dist / lastTouchDist.current
   140→   140→      setZoom(z => {
   141→   141→        const next = Math.min(Math.max(z * scale, 0.5), 5)
   142→   142→        if (next <= 1) setPosition({ x: 0, y: 0 })
   143→   143→        return next
   144→   144→      })
   145→   145→      lastTouchDist.current = dist
   146→   146→    } else if (e.touches.length === 1 && isDragging) {
   147→   147→      setPosition({
   148→   148→        x: posStartRef.current.x + (e.touches[0].clientX - dragStartRef.current.x),
   149→   149→        y: posStartRef.current.y + (e.touches[0].clientY - dragStartRef.current.y),
   150→   150→      })
   151→   151→    }
   152→   152→  }
   153→   153→
   154→   154→  const handleTouchEnd = () => {
   155→   155→    lastTouchDist.current = null
   156→   156→    setIsDragging(false)
   157→   157→  }
   158→   158→
   159→   159→
   160→   160→  // Close lightbox when clicking the dark overlay area (not the image or controls)
   161→   161→  const handleOverlayClick = useCallback((e: React.MouseEvent) => {
   162→   162→    // Only close if clicking directly on the overlay, not on children
   163→   163→    if (e.target === e.currentTarget) {
   164→   164→      onClose()
   165→   165→    }
   166→   166→  }, [onClose])
   167→   167→
   168→   168→  if (!isOpen) return null
   169→   169→
   170→   170→  const lightboxContent = (
   171→   171→    <div
   172→   172→      className="fixed inset-0 z-[9999] bg-black/90 flex flex-col items-center justify-center pointer-events-auto"
   173→   173→      onClick={handleOverlayClick}
   174→   174→    >
   175→   175→      {/* Close button */}
   176→   176→      <button
   177→   177→        onClick={(e) => { e.stopPropagation(); onClose() }}
   178→   178→        className="absolute top-4 right-4 z-10 text-white/70 hover:text-white transition-colors p-2 rounded-full hover:bg-white/10"
   179→   179→        aria-label="Fermer"
   180→   180→      >
   181→   181→        <X className="size-6" />
   182→   182→      </button>
   183→   183→      <div
   184→   184→        ref={containerRef}
   185→   185→        className="flex-1 w-full flex items-center justify-center overflow-hidden"
   186→   186→        onWheel={handleWheel}
   187→   187→        onPointerDown={handlePointerDown}
   188→   188→        onTouchStart={handleTouchStart}
   189→   189→        onTouchMove={handleTouchMove}
   190→   190→        onTouchEnd={handleTouchEnd}
   191→   191→        style={{ cursor: zoom > 1 ? (isDragging ? 'grabbing' : 'grab') : 'default', userSelect: 'none' }}
   192→   192→      >
   193→   193→        {!imageLoaded && (
   194→   194→          <Loader2 className="size-10 text-white animate-spin" />
   195→   195→        )}
   196→   196→        <img
   197→   197→          src={imageUrl}
   198→   198→          alt={label}
   199→   199→          className="max-w-[90vw] max-h-[80vh] object-contain select-none pointer-events-none"
   200→   200→          style={{
   201→   201→            transform: `scale(${zoom}) translate(${position.x / zoom}px, ${position.y / zoom}px)`,
   202→   202→            transition: isDragging ? 'none' : 'transform 0.15s ease',
   203→   203→            display: imageLoaded ? 'block' : 'none',
   204→   204→          }}
   205→   205→          onLoad={() => setImageLoaded(true)}
   206→   206→          draggable={false}
   207→   207→        />
   208→   208→      </div>
   209→   209→      {/* Zoom control bar */}
   210→   210→      <div
   211→   211→        className="flex items-center gap-4 py-4 px-6 bg-black/70 rounded-t-xl"
   212→   212→        onClick={(e) => e.stopPropagation()}
   213→   213→      >
   214→   214→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.max(z - 0.25, 0.5)) }}>
   215→   215→          <ZoomOut className="size-5" />
   216→   216→        </Button>
   217→   217→        <span className="text-white text-sm font-mono min-w-[50px] text-center">{Math.round(zoom * 100)}%</span>
   218→   218→        <Button variant="ghost" size="icon" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); setZoom(z => Math.min(z + 0.25, 5)) }}>
   219→   219→          <ZoomIn className="size-5" />
   220→   220→        </Button>
   221→   221→        <Button variant="ghost" size="sm" className="text-white hover:bg-white/20" onClick={(e) => { e.stopPropagation(); resetView() }}>
   222→   222→          <RotateCcw className="size-4 mr-1" /> Réinitialiser
   223→   223→        </Button>
   224→   224→        <span className="text-white/70 text-sm ml-2">{label}</span>
   225→   225→      </div>
   226→   226→    </div>
   227→   227→  )
   228→   228→
   229→   229→  // Render via portal to document.body so it sits at the same DOM level as the
   230→   230→  // Radix Dialog portal and is guaranteed to be on top (z-9999 > dialog z-50).
   231→   231→  if (typeof document === 'undefined') return null
   232→   232→  return createPortal(lightboxContent, document.body)
   233→   233→}

   235→   235→// ===== Exercise Detail Dialog =====
   236→   236→function ExerciseDetailDialog({
   237→   237→  exercise,
   238→   238→  chapter,
   239→   239→  exercises,
   240→   240→  onClose,
   241→   241→}: {
   242→   242→  exercise: ExerciseData
   243→   243→  chapter: ChapterData
   244→   244→  exercises: ExerciseData[]
   245→   245→  onClose: () => void
   246→   246→}) {
   247→   247→  const { user, setSelectedExercise, openLightbox, lightbox } = useAppStore()
   248→   248→  const student = user as StudentUser
   249→   249→  const [status, setStatus] = useState(exercise.progress?.status || 'not_started')
   250→   250→  const [note, setNote] = useState(exercise.progress?.studentNote || '')
   251→   251→  const [saving, setSaving] = useState(false)
   252→   252→  const [comments, setComments] = useState<CommentData[]>([])
   253→   253→  const [newComment, setNewComment] = useState('')
   254→   254→  const [sendingComment, setSendingComment] = useState(false)
   255→   255→
   256→   256→  const currentIndex = exercises.findIndex(e => e.id === exercise.id)
   257→   257→  const totalExercises = exercises.length
   258→   258→
   259→   259→  const [commentsLoaded, setCommentsLoaded] = useState(false)
   260→   260→
   261→   261→  useEffect(() => {
   262→   262→    let cancelled = false
   263→   263→    if (commentsLoaded) return
   264→   264→    const load = async () => {
   265→   265→      try {
   266→   266→        const res = await fetch('/api/comments')
   267→   267→        if (res.ok && !cancelled) {
   268→   268→          const data = await res.json()
   269→   269→          setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   270→   270→        }
   271→   271→      } catch { /* ignore */ }
   272→   272→      if (!cancelled) setCommentsLoaded(true)
   273→   273→    }
   274→   274→    load()
   275→   275→  }, [exercise.id, commentsLoaded])
   276→   276→
   277→   277→  const refetchComments = async () => {
   278→   278→    try {
   279→   279→      const res = await fetch('/api/comments')
   280→   280→      if (res.ok) {
   281→   281→        const data = await res.json()
   282→   282→        setComments((data.comments || []).filter((c: CommentData) => c.exercise?.id === exercise.id))
   283→   283→      }
   284→   284→    } catch { /* ignore */ }
   285→   285→  }
   286→   286→
   287→   287→  const handleSave = async () => {
   288→   288→    setSaving(true)
   289→   289→    try {
   290→   290→      await fetch(`/api/exercises/${exercise.id}/progress`, {
   291→   291→        method: 'PUT',
   292→   292→        headers: { 'Content-Type': 'application/json' },
   293→   293→        body: JSON.stringify({ studentId: student.id, status, studentNote: note }),
   294→   294→      })
   295→   295→      toast.success('Progression sauvegardée !')
   296→   296→    } catch {
   297→   297→      toast.error('Erreur lors de la sauvegarde.')
   298→   298→    }
   299→   299→    setSaving(false)
   300→   300→  }
   301→   301→
   302→   302→  const handleSendComment = async () => {
   303→   303→    if (!newComment.trim()) return
   304→   304→    setSendingComment(true)
   305→   305→    try {
   306→   306→      await fetch('/api/comments', {
   307→   307→        method: 'POST',
   308→   308→        headers: { 'Content-Type': 'application/json' },
   309→   309→        body: JSON.stringify({ studentId: student.id, exerciseId: exercise.id, content: newComment }),
   310→   310→      })
   311→   311→      setNewComment('')
   312→   312→      refetchComments()
   313→   313→    } catch {
   314→   314→      toast.error("Erreur lors de l'envoi.")
   315→   315→    }
   316→   316→    setSendingComment(false)
   317→   317→  }
   318→   318→
   319→   319→  const goToExercise = (direction: 'prev' | 'next') => {
   320→   320→    const newIndex = direction === 'prev' ? currentIndex - 1 : currentIndex + 1
   321→   321→    if (newIndex < 0 || newIndex >= totalExercises) return
   322→   322→    setSelectedExercise(exercises[newIndex])
   323→   323→  }
   324→   324→
   325→   325→  // Build list of page images for this exercise
   326→   326→  const pageImages: { url: string; pageNum: number }[] = []
   327→   327→  if (exercise.pageStart) {
   328→   328→    const start = exercise.pageStart
   329→   329→    const end = exercise.pageEnd || start
   330→   330→    for (let p = start; p <= end; p++) {
   331→   331→      pageImages.push({ url: pageImageUrl(p), pageNum: p })
   332→   332→    }
   333→   333→  }
   334→   334→
   335→   335→  return (
   336→   336→      <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>
   337→   337→        <DialogContent 
   338→   338→          className="max-w-4xl max-h-[90vh] overflow-y-auto"
   339→   339→          onInteractOutside={(e) => { if (lightbox.isOpen) e.preventDefault() }}
   340→   340→        >
   341→   341→          <DialogHeader>
   342→   342→            <DialogTitle className="text-lg">
   343→   343→              Ch.{chapter.number} : {chapter.title} — Ex.{exercise.number} ({exercise.number}/{totalExercises})
   344→   344→            </DialogTitle>
   345→   345→            <DialogDescription className="sr-only">
   346→   346→              Détail de l&apos;exercice {exercise.number} du chapitre {chapter.number}
   347→   347→            </DialogDescription>
   348→   348→          </DialogHeader>
   349→   349→
   350→   350→          {pageImages.length > 0 && (
   351→   351→            <div className="space-y-3">
   352→   352→              {pageImages.map((img) => (
   353→   353→                <div key={img.pageNum} className="space-y-1.5">
   354→   354→                  <div
   355→   355→                    className="relative rounded-lg overflow-hidden border bg-white cursor-pointer"
   356→   356→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   357→   357→                  >
   358→   358→                    <img
   359→   359→                      src={img.url}
   360→   360→                      alt={`Page ${img.pageNum} — Exercice ${exercise.number}`}
   361→   361→                      className="w-full h-auto object-contain"
   362→   362→                      loading="lazy"
   363→   363→                    />
   364→   364→                  </div>
   365→   365→                  <Button
   366→   366→                    variant="outline"
   367→   367→                    size="sm"
   368→   368→                    className="w-full"
   369→   369→                    onClick={() => openLightbox(img.url, `Page ${img.pageNum}`)}
   370→   370→                  >
   371→   371→                    <Eye className="size-4 mr-1" /> Agrandir la page {img.pageNum}
   372→   372→                  </Button>
   373→   373→                </div>
   374→   374→              ))}
   375→   375→            </div>
   376→   376→          )}
   377→   377→
   378→   378→          <div className="space-y-2">
   379→   379→            <Label className="text-sm font-medium">Statut</Label>
   380→   380→            <RadioGroup value={status} onValueChange={setStatus} className="grid grid-cols-2 gap-2">
   381→   381→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-gray-400">
   382→   382→                <RadioGroupItem value="not_started" />
   383→   383→                <Circle className="size-4 text-gray-500" />
   384→   384→                <span className="text-sm text-gray-500">Non commencé</span>
   385→   385→              </label>
   386→   386→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-amber-400">
   387→   387→                <RadioGroupItem value="in_progress" />
   388→   388→                <Clock className="size-4 text-amber-600" />
   389→   389→                <span className="text-sm text-amber-700">En cours</span>
   390→   390→              </label>
   391→   391→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-emerald-400">
   392→   392→                <RadioGroupItem value="completed" />
   393→   393→                <CheckCircle className="size-4 text-emerald-600" />
   394→   394→                <span className="text-sm text-emerald-700">Terminé</span>
   395→   395→              </label>
   396→   396→              <label className="flex items-center gap-2 cursor-pointer rounded-lg border p-2 hover:bg-gray-50 transition-colors has-[button[data-state=checked]]:border-red-400">
   397→   397→                <RadioGroupItem value="difficulty" />
   398→   398→                <AlertTriangle className="size-4 text-red-600" />
   399→   399→                <span className="text-sm text-red-700">Difficulté</span>
   400→   400→              </label>
   401→   401→            </RadioGroup>
   402→   402→          </div>
   403→   403→
   404→   404→          <div className="space-y-2">
   405→   405→            <Label htmlFor="notes" className="text-sm font-medium">Notes personnelles</Label>
   406→   406→            <Textarea
   407→   407→              id="notes"
   408→   408→              value={note}
   409→   409→              onChange={(e) => setNote(e.target.value)}
   410→   410→              placeholder="Ajoutez vos notes ici..."
   411→   411→              rows={3}
   412→   412→            />
   413→   413→          </div>
   414→   414→
   415→   415→          <div className="flex justify-end">
   416→   416→            <Button onClick={handleSave} disabled={saving} className="bg-emerald-600 hover:bg-emerald-700">
   417→   417→              {saving && <Loader2 className="size-4 mr-2 animate-spin" />}
   418→   418→              Sauvegarder
   419→   419→            </Button>
