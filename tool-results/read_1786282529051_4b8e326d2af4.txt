  1405→          </div>
  1406→

  1408→            <div className="space-y-2 mb-3">
  1409→              {comment.remarks.map((remark) => (

  1449→function TeacherProfileView() {
  1450→  const { user, logout } = useAppStore()
  1451→  const teacher = user as TeacherUser
  1452→  const [profile, setProfile] = useState<TeacherUser | null>(null)
  1453→  const [currentPassword, setCurrentPassword] = useState('')
  1454→  const [newPassword, setNewPassword] = useState('')
  1455→  const [confirmPassword, setConfirmPassword] = useState('')
  1456→  const [saving, setSaving] = useState(false)
  1457→  const [error, setError] = useState('')
  1458→
  1459→  useEffect(() => {
  1460→    const fetchProfile = async () => {
  1461→      try {
  1462→        const res = await fetch('/api/teacher/profile')
  1463→        if (res.ok) {
  1464→          const data = await res.json()
  1465→          setProfile(data)
  1466→        }
  1467→      } catch { /* ignore */ }
  1468→    }
  1469→    fetchProfile()
  1470→  }, [])
  1471→
  1472→  const displayTeacher = profile || teacher
  1473→
  1474→  const handleChangePassword = async () => {
  1475→    setError('')
  1476→    if (!currentPassword || !newPassword || !confirmPassword) {
  1477→      setError('Tous les champs sont requis.')
  1478→      return
  1479→    }

  1481→      setError('Le nouveau mot de passe doit contenir au moins 4 caractères.')

  1483→    }

  1485→      setError('Les mots de passe ne correspondent pas.')
  1486→      return
  1487→    }
  1488→    setSaving(true)
  1489→    try {
  1490→      const res = await fetch('/api/teacher/profile', {
  1491→        method: 'PUT',
  1492→        headers: { 'Content-Type': 'application/json' },
  1493→        body: JSON.stringify({ currentPassword, newPassword }),
  1494→      })
  1495→      if (res.ok) {
  1496→        toast.success('Mot de passe modifié.')
  1497→        setCurrentPassword('')
  1498→        setNewPassword('')
  1499→        setConfirmPassword('')
  1500→      } else {
  1501→        setError('Mot de passe actuel incorrect.')
  1502→      }
  1503→    } catch {
  1504→      setError('Erreur lors de la modification.')
  1505→    }
  1506→    setSaving(false)
  1507→  }
  1508→
  1509→  return (
  1510→    <div className="max-w-md mx-auto space-y-6">
  1511→      <Card>
  1512→        <CardHeader className="flex flex-col items-center text-center">
  1513→          <Avatar className="size-16">

  1516→            </AvatarFallback>

  1519→          <CardDescription>{displayTeacher.status}</CardDescription>
  1520→        </CardHeader>
  1521→        <CardContent>
  1522→          <div className="flex items-center gap-2 text-sm justify-center">
  1523→            <Mail className="size-4 text-muted-foreground" />

  1529→      <Card>
